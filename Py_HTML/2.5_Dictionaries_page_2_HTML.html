Time-Series Vital Signs

Patient Vital Signs Throughout the Day 
# Patient vital signs throughout the day
vital_signs_log = {
    "06:00": {"temp": 98.2, "pulse": 68, "bp": "118/76", "o2_sat": 98},
    "10:00": {"temp": 98.6, "pulse": 72, "bp": "120/80", "o2_sat": 97},
    "14:00": {"temp": 99.1, "pulse": 78, "bp": "125/82", "o2_sat": 97},
    "18:00": {"temp": 99.8, "pulse": 84, "bp": "130/85", "o2_sat": 96},
    "22:00": {"temp": 100.2, "pulse": 88, "bp": "132/86", "o2_sat": 95}
}

# Analyze temperature trend
print("🌡️ Temperature Trend Analysis:")
temps = []
for time, vitals in vital_signs_log.items():
    temp = vitals["temp"]
    temps.append(temp)
    status = "⚠️ FEVER" if temp >= 100.4 else "📈 Elevated" if temp > 99 else "✅ Normal"
    print(f"  {time}: {temp}°F {status}")

# Calculate statistics
avg_temp = sum(temps) / len(temps)
max_temp = max(temps)
print(f"\nAverage: {avg_temp:.1f}°F")
print(f"Peak: {max_temp}°F at ", end="")
for time, vitals in vital_signs_log.items():
    if vitals["temp"] == max_temp:
        print(time)

print("\n🚨 Clinical Alerts:")
alerts_found = False

    alerts = []
    
    if vitals["temp"] >= 100.4:
        alerts.append("Fever")

Complete solution with comprehensive checking 
current_medications = {
    "warfarin": {"dose": "5mg", "class": "anticoagulant"},
    "metoprolol": {"dose": "50mg", "class": "beta-blocker"},
    "simvastatin": {"dose": "40mg", "class": "statin"}
}

interaction_rules = {
    ("anticoagulant", "antiplatelet"): "High bleeding risk",
    ("beta-blocker", "calcium-channel-blocker"): "Risk of bradycardia",
    ("statin", "fibrate"): "Increased myopathy risk"
}

new_drug = "aspirin"
new_drug_class = "antiplatelet"

print(f"💊 Checking interactions for {new_drug} ({new_drug_class})...")
print("=" * 50)

interactions_found = []

# Check each current medication
for med_name, med_info in current_medications.items():
    current_class = med_info["class"]
    
    # Check both directions of interaction
    if (current_class, new_drug_class) in interaction_rules:
        risk = interaction_rules[(current_class, new_drug_class)]
        interactions_found.append({
            "drugs": f"{new_drug} + {med_name}",
            "risk": risk,
            "severity": "HIGH" if "High" in risk else "MODERATE"
        })
    elif (new_drug_class, current_class) in interaction_rules:
        risk = interaction_rules[(new_drug_class, current_class)]
        interactions_found.append({
            "drugs": f"{new_drug} + {med_name}", 
            "risk": risk,
            "severity": "HIGH" if "High" in risk else "MODERATE"
        })

# Report findings
if interactions_found:
    print("\n⚠️ DRUG INTERACTIONS DETECTED!")
    for interaction in interactions_found:
        severity_emoji = "🔴" if interaction["severity"] == "HIGH" else "🟡"
        print(f"\n{severity_emoji} {interaction['severity']} RISK")
        print(f"   Drugs: {interaction['drugs']}")
        print(f"   Risk: {interaction['risk']}")
        print(f"   Action: Consult physician before prescribing")
else:
    print("\n✅ No major interactions detected")
    print(f"   {new_drug} appears safe with current medications")

# Calculate treatment effectiveness
experimental_improvements = []
control_improvements = []

for patient in clinical_trial:
    hba1c_change = patient["baseline_hba1c"] - patient["week_12_hba1c"]
    weight_change = patient["baseline_weight"] - patient["week_12_weight"]
    
    result = {
        "id": patient["patient_id"],
        "hba1c_improvement": hba1c_change,
        "weight_loss": weight_change
    }
    
    if patient["treatment_group"] == "experimental":
        experimental_improvements.append(hba1c_change)
    else:
        control_improvements.append(hba1c_change)

# Calculate averages

Complete solution - Population Health Dashboard 
print("🏥 DIABETES POPULATION HEALTH DASHBOARD")
print("=" * 50)

# Task 1: Poor glycemic control
poor_control = []
for patient in diabetes_registry:
    if patient["hba1c"] > 8.0:
        poor_control.append(patient)

print(f"\n📊 Glycemic Control Analysis:")
print(f"  Total patients: {len(diabetes_registry)}")
print(f"  Poor control (HbA1c >8): {len(poor_control)} ({len(poor_control)/len(diabetes_registry)*100:.0f}%)")

# Task 2: Multiple complications
high_risk_patients = []
for patient in diabetes_registry:
    if len(patient["complications"]) >= 2:
        high_risk_patients.append(patient)

print(f"\n⚠️ High-Risk Patients (2+ complications): {len(high_risk_patients)}")
for patient in high_risk_patients:
    complications_list = ", ".join(patient["complications"])
    print(f"  • {patient['id']}: HbA1c {patient['hba1c']}, Complications: {complications_list}")

# Task 3: BMI by medication
med_groups = {}
for patient in diabetes_registry:
    med = patient["medication"]
    if med not in med_groups:
        med_groups[med] = []
    med_groups[med].append(patient["bmi"])

print(f"\n💊 Average BMI by Medication:")
for med, bmi_list in med_groups.items():
    avg_bmi = sum(bmi_list) / len(bmi_list)
    print(f"  • {med.capitalize()}: {avg_bmi:.1f} (n={len(bmi_list)})")

# Task 4: Generate recommendations
print(f"\n📋 PERSONALIZED RECOMMENDATIONS:")
print("-" * 50)

for patient in diabetes_registry:
    if patient["hba1c"] > 9.0 or len(patient["complications"]) >= 2:
        print(f"\nPatient {patient['id']}:")
        
        # Medication recommendation
        if patient["hba1c"] > 9.0 and patient["medication"] != "insulin":
            print(f"  💉 Consider insulin therapy (HbA1c: {patient['hba1c']})")
        
        # Weight management
        if patient["bmi"] > 30:
            weight_loss_target = (patient["bmi"] - 25) * 2.2 * 70 / 100  # rough estimate
            print(f"  🏃 Weight loss goal: {weight_loss_target:.0f} lbs (BMI: {patient['bmi']})")
        
        # Complication screening
        all_complications = ["retinopathy", "neuropathy", "nephropathy"]
        missing_screens = [c for c in all_complications if c not in patient["complications"]]
        if missing_screens and patient["hba1c"] > 8.0:
            print(f"  👁️ Screen for: {', '.join(missing_screens)}")
        
        # Specialist referral
        if len(patient["complications"]) >= 2:
            print(f"  👨‍⚕️ Refer to endocrinologist for intensive management")

# Summary statistics
print(f"\n📈 POPULATION SUMMARY:")
all_hba1c = [p["hba1c"] for p in diabetes_registry]
print(f"  • Average HbA1c: {sum(all_hba1c)/len(all_hba1c):.1f}%")
print(f"  • Patients at goal (<7%): {len([p for p in diabetes_registry if p['hba1c'] < 7])} ({len([p for p in diabetes_registry if p['hba1c'] < 7])/len(diabetes_registry)*100:.0f}%)")
print(f"  • Insulin users: {len([p for p in diabetes_registry if p['medication'] == 'insulin'])}")
