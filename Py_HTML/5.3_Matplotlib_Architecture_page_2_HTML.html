MODULE 2: RC Parameters and Professional Medical Visualization Standards

🔬 Mission Briefing: RC Parameters - Your Hospital Style Guide 📋

Remember how every hospital has its own way of doing things? RC (Runtime Configuration) parameters are Matplotlib's style guide - they ensure every chart in your hospital looks consistent and professional!

Think of it as setting hospital-wide standards:

  • "All temperature charts must use Celsius"
  • "Critical values must be in red"
  • "Patient names go in the top-left corner"

🧪 Creating Your Hospital's Visual Standards

# First, let's see the current "hospital standards"
import matplotlib.pyplot as plt
import numpy as np

print("🏥 Current Hospital Chart Standards:")
print(f"Default chart size: {plt.rcParams['figure.figsize']} inches")
print(f"Default font size: {plt.rcParams['font.size']} points")
print(f"Default line width: {plt.rcParams['lines.linewidth']} points")
print(f"Grid enabled: {plt.rcParams['axes.grid']}")

# Now let's create professional medical standards!
# (Like upgrading from hand-drawn charts to digital ones!)

medical_style = {
    # Chart sizing - bigger for better readability in clinical settings
    'figure.figsize': (12, 8),
    'figure.dpi': 100,
    
    # Typography - clear and professional
    'font.family': 'DejaVu Sans',  # Clear, readable font
    'font.size': 11,
    'axes.labelsize': 12,
    'axes.titlesize': 14,
    'legend.fontsize': 10,
    
    # Lines and markers - visible from across the room!
    'lines.linewidth': 2.5,
    'lines.markersize': 8,
    'lines.markeredgewidth': 1.5,
    
    # Grid - helps read values quickly
    'axes.grid': True,
    'grid.alpha': 0.3,
    'grid.linestyle': ':',
    
    # Colors - colorblind-friendly palette
    'axes.prop_cycle': plt.cycler('color', 
        ['#0173B2', '#DE8F05', '#029E73', '#CC78BC', '#CA9161',
         '#FBAFE4', '#949494', '#ECE133', '#56B4E9']),
    
    # Professional appearance
    'figure.facecolor': 'white',
    'axes.facecolor': '#F8F8F8',
    'axes.edgecolor': '#CCCCCC',
    'axes.linewidth': 1.0,
    
    # Save settings for reports
    'savefig.dpi': 300,  # High quality for printing
    'savefig.bbox': 'tight',
    'savefig.pad_inches': 0.2
}

# Apply our medical standards
plt.rcParams.update(medical_style)

# Test our new style with vital signs data!
time = np.linspace(0, 10, 50)
heart_rate = 70 + 15 * np.sin(time) + np.random.normal(0, 2, len(time))
resp_rate = 16 + 4 * np.sin(time * 1.5) + np.random.normal(0, 1, len(time))

fig, (ax1, ax2) = plt.subplots(2, 1, sharex=True)

# Heart rate plot
ax1.plot(time, heart_rate, 'o-', label='Heart Rate', markersize=4)
ax1.axhspan(60, 100, alpha=0.2, color='green', label='Normal Range')
ax1.set_ylabel('Heart Rate (BPM)')
ax1.set_title('🏥 Patient Vital Signs - Medical Style Applied!', pad=20)
ax1.legend(loc='upper right')

# Respiratory rate plot
ax2.plot(time, resp_rate, 's-', label='Respiratory Rate', markersize=4)
ax2.axhspan(12, 20, alpha=0.2, color='green', label='Normal Range')
ax2.set_xlabel('Time (hours)')
ax2.set_ylabel('Resp. Rate (breaths/min)')
ax2.legend(loc='upper right')

# Add professional timestamp
fig.text(0.99, 0.01, f'Generated: {np.datetime64("now")}', 
         ha='right', va='bottom', fontsize=9, alpha=0.6)

plt.tight_layout()
plt.show()

# Don't forget to reset if needed!
# plt.rcdefaults()

# St. Luke's Medical Center Professional Chart Template
import matplotlib.pyplot as plt
import numpy as np
from datetime import datetime

# Hospital-approved configuration
st_lukes_config = {
    # Size and resolution
    'figure.figsize': (14, 8),
    'figure.dpi': 100,
    
    # Typography - AMA style guide compliant
    'font.family': 'serif',
    'font.size': 13,
    'axes.titlesize': 18,
    'axes.titleweight': 'bold',
    'axes.labelsize': 14,
    'axes.labelweight': 'bold',
    'legend.fontsize': 11,
    'xtick.labelsize': 11,
    'ytick.labelsize': 11,
    
    # Lines and markers - high visibility
    'lines.linewidth': 2.8,
    'lines.markersize': 9,
    'lines.markeredgewidth': 2,
    'lines.markeredgecolor': 'white',
    
    # Grid - subtle but helpful
    'axes.grid': True,
    'grid.alpha': 0.35,
    'grid.color': '#CCCCCC',
    'grid.linestyle': ':',
    
    # Colorblind-safe palette (validated by ophthalmology dept!)
    'axes.prop_cycle': plt.cycler('color', 
        ['#E74C3C',  # Red - Critical
         '#3498DB',  # Blue - Primary
         '#2ECC71',  # Green - Normal
         '#F39C12',  # Orange - Warning
         '#9B59B6',  # Purple - Secondary
         '#1ABC9C',  # Teal - Tertiary
         '#34495E']), # Dark Gray - Reference
    
    # Professional appearance
    'figure.facecolor': 'white',
    'axes.facecolor': '#FCFCFC',
    'axes.edgecolor': '#333333',
    'axes.linewidth': 1.2,
    'axes.spines.top': False,
    'axes.spines.right': False,
    
    # Legend styling
    'legend.frameon': True,
    'legend.fancybox': True,
    'legend.shadow': True,
    'legend.framealpha': 0.9,
    'legend.edgecolor': '#CCCCCC',
    
    # Export settings for medical records
    'savefig.dpi': 300,
    'savefig.bbox': 'tight',
    'savefig.pad_inches': 0.3,
    'savefig.facecolor': 'white',
    'savefig.edgecolor': 'none'
}

# Apply St. Luke's standards
plt.rcParams.update(st_lukes_config)

# Create comprehensive patient monitoring dashboard
fig = plt.figure(figsize=(14, 10))

# Create grid for multiple plots
gs = fig.add_gridspec(3, 2, height_ratios=[2, 2, 1], hspace=0.3, wspace=0.25)

# Generate realistic patient data
hours = np.linspace(0, 24, 100)
heart_rate = 75 + 10 * np.sin(hours/4) + np.random.normal(0, 3, 100)
blood_pressure_sys = 120 + 8 * np.sin(hours/5) + np.random.normal(0, 4, 100)
blood_pressure_dia = 80 + 5 * np.sin(hours/5) + np.random.normal(0, 3, 100)
temperature = 37 + 0.5 * np.sin(hours/6) + np.random.normal(0, 0.1, 100)
oxygen_sat = 97 + 2 * np.sin(hours/3) + np.random.normal(0, 1, 100)

# Plot 1: Heart Rate
ax1 = fig.add_subplot(gs[0, 0])
ax1.plot(hours, heart_rate, '-', label='Heart Rate', linewidth=2)
ax1.axhspan(60, 100, alpha=0.2, color='green', label='Normal Range')
ax1.set_ylabel('Heart Rate (BPM)')
ax1.set_title('Cardiac Monitoring')
ax1.legend(loc='upper right')
ax1.set_ylim(40, 120)

# Plot 2: Blood Pressure
ax2 = fig.add_subplot(gs[0, 1])
ax2.plot(hours, blood_pressure_sys, '-', label='Systolic', linewidth=2)
ax2.plot(hours, blood_pressure_dia, '-', label='Diastolic', linewidth=2)
ax2.axhspan(90, 120, alpha=0.2, color='green')
ax2.axhspan(60, 80, alpha=0.2, color='green')
ax2.set_ylabel('Blood Pressure (mmHg)')
ax2.set_title('Blood Pressure Trends')
ax2.legend(loc='upper right')
ax2.set_ylim(40, 160)

# Plot 3: Temperature
ax3 = fig.add_subplot(gs[1, 0])
ax3.plot(hours, temperature, '-', label='Core Temperature', linewidth=2, color='#E74C3C')
ax3.axhspan(36.5, 37.5, alpha=0.2, color='green', label='Normal Range')
ax3.set_ylabel('Temperature (°C)')
ax3.set_title('Temperature Monitoring')
ax3.legend(loc='upper right')
ax3.set_ylim(35.5, 38.5)

# Plot 4: Oxygen Saturation
ax4 = fig.add_subplot(gs[1, 1])
ax4.plot(hours, oxygen_sat, '-', label='SpO₂', linewidth=2, color='#3498DB')
ax4.axhspan(95, 100, alpha=0.2, color='green', label='Normal Range')
ax4.axhline(90, color='red', linestyle='--', alpha=0.7, label='Critical Threshold')
ax4.set_ylabel('Oxygen Saturation (%)')
ax4.set_title('Pulse Oximetry')
ax4.legend(loc='lower right')
ax4.set_ylim(85, 102)

# Set common x-label for bottom plots
ax3.set_xlabel('Time (hours)')
ax4.set_xlabel('Time (hours)')

# Patient Information Panel
ax5 = fig.add_subplot(gs[2, :])
ax5.axis('off')

# Create patient info text
patient_info = f"""
🏥 ST. LUKE'S MEDICAL CENTER - PATIENT MONITORING DASHBOARD

👤 Patient: John Smith | MRN: 12345678 | Room: ICU-7
📋 Diagnosis: Post-operative monitoring (Cardiac surgery)
💊 Current Medications: Metoprolol 50mg bid, Aspirin 81mg daily, Atorvastatin 40mg daily
🩺 Attending: Dr. Sarah Johnson | Last Updated: {datetime.now().strftime('%Y-%m-%d %H:%M:%S')}

⚠️ ALERTS: None | 📊 Monitoring Duration: 24 hours | 🔄 Auto-refresh: Every 5 minutes
"""

ax5.text(0.5, 0.5, patient_info, transform=ax5.transAxes, 
         fontsize=11, ha='center', va='center',
         bbox=dict(boxstyle='round,pad=1', facecolor='#E8F4F8', 
                   edgecolor='#3498DB', linewidth=2))

# Add hospital logo/branding
fig.text(0.02, 0.98, '🏥 St. Luke\'s Medical Center', 
         transform=fig.transFigure, fontsize=20, 
         weight='bold', va='top', color='#2C3E50')

# Add timestamp and quality indicators
fig.text(0.98, 0.02, 
         'Quality Metrics: Signal Quality ✓ | Data Integrity ✓ | Calibrated ✓', 
         transform=fig.transFigure, fontsize=9, 
         ha='right', alpha=0.7, style='italic')

plt.suptitle('Comprehensive Patient Monitoring Report', 
             fontsize=22, y=0.96, weight='bold')

plt.tight_layout()
plt.show()

# Save a sample (this would go to the medical records system)
plt.savefig('patient_monitoring_report.pdf', format='pdf')

print("\n✅ St. Luke's template applied successfully!")
print("📊 This template is now the standard for all departments.")
