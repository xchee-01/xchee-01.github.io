cholesterol_levels = np.array([185, 220, 195, 240, 178, 205, 198, 215, 188, 225])

print(f"🧪 Analyzing {len(cholesterol_levels)} patient cholesterol samples...")

📊 Calculating Key Statistics

# Calculate the statistical story of our patient cohort
mean_chol = np.mean(cholesterol_levels)
std_chol = np.std(cholesterol_levels)
median_chol = np.median(cholesterol_levels)

# Percentiles tell us about distribution
percentile_25 = np.percentile(cholesterol_levels, 25)
percentile_75 = np.percentile(cholesterol_levels, 75)

print(f"\n📊 Statistical Summary:")
print(f"Mean: {mean_chol:.1f} mg/dL")
print(f"Median: {median_chol:.1f} mg/dL")
print(f"Std Dev: {std_chol:.1f} mg/dL")
print(f"IQR: {percentile_25:.1f} - {percentile_75:.1f} mg/dL")

🚨 Risk Stratification

# Identify patients needing intervention (>200 mg/dL is elevated)
high_risk = cholesterol_levels > 200
borderline = (cholesterol_levels >= 180) & (cholesterol_levels <= 200)
optimal = cholesterol_levels < 180

# Create a risk report
risk_percentage = (np.sum(high_risk) / len(cholesterol_levels)) * 100

print(f"\n🚦 Risk Stratification:")
print(f"🟢 Optimal (<180): {np.sum(optimal)} patients")
print(f"🟡 Borderline (180-200): {np.sum(borderline)} patients")
print(f"🔴 High risk (>200): {np.sum(high_risk)} patients ({risk_percentage:.0f}%)")

# Statistical significance check
if risk_percentage > 50:
    print("\n⚠️ Alert: Majority of cohort at elevated risk!")
    print("📋 Recommendation: Implement clinic-wide lifestyle intervention program")

Complete Clinical Trial Analysis 

# Complete clinical trial analysis with FDA-style reporting! 📊
baseline_bp = np.array([145, 152, 138, 160, 142, 155, 148, 162, 140, 158])
post_treatment_bp = np.array([128, 135, 130, 142, 125, 138, 132, 145, 124, 140])

print("🧪 CLINICAL TRIAL ANALYSIS REPORT")
print("=" * 50)
print(f"Trial Size: {len(baseline_bp)} patients")
print(f"Treatment Duration: 4 weeks")

# Calculate treatment effects
absolute_reduction = baseline_bp - post_treatment_bp
percent_reduction = (absolute_reduction / baseline_bp) * 100

# Identify responders (>10% reduction is clinically meaningful)
responders = percent_reduction > 10
response_rate = (np.sum(responders) / len(baseline_bp)) * 100

# Statistical summary
mean_reduction = np.mean(absolute_reduction)
std_reduction = np.std(absolute_reduction)
median_reduction = np.median(absolute_reduction)

# Effect size calculation (Cohen's d)
effect_size = mean_reduction / std_reduction

print(f"\n📊 PRIMARY OUTCOMES:")
print(f"Mean BP reduction: {mean_reduction:.1f} ± {std_reduction:.1f} mmHg")
print(f"Median reduction: {median_reduction:.1f} mmHg")
print(f"Range: {np.min(absolute_reduction):.0f} - {np.max(absolute_reduction):.0f} mmHg")
print(f"Effect size (Cohen's d): {effect_size:.2f} {'(Large)' if effect_size > 0.8 else '(Medium)'}")

print(f"\n📈 RESPONSE ANALYSIS:")
print(f"Treatment responders: {np.sum(responders)}/{len(baseline_bp)} ({response_rate:.0f}%)")
print(f"FDA threshold: 60% (Requirement {'MET ✅' if response_rate >= 60 else 'NOT MET ❌'})")

# Individual patient outcomes
print(f"\n👥 PATIENT-LEVEL RESULTS:")
for i in range(len(baseline_bp)):
    emoji = "✅" if responders[i] else "➖"
    print(f"Patient {i+1}: {baseline_bp[i]} → {post_treatment_bp[i]} "
          f"(-{absolute_reduction[i]} mmHg, -{percent_reduction[i]:.1f}%) {emoji}")

# Clinical significance analysis
clinically_significant = absolute_reduction >= 10
print(f"\n🏥 CLINICAL SIGNIFICANCE:")
print(f"Clinically significant responses (≥10 mmHg): {np.sum(clinically_significant)}/{len(baseline_bp)}")

# Safety analysis
excessive_reduction = percent_reduction > 20
print(f"\n⚠️ SAFETY MONITORING:")
print(f"Patients with >20% reduction: {np.sum(excessive_reduction)}")
if np.sum(excessive_reduction) > 0:
    print("Note: Monitor these patients for hypotension symptoms")

# FDA recommendation
print(f"\n📋 RECOMMENDATION:")
if response_rate >= 60 and mean_reduction >= 10:
    print("✅ APPROVE: Treatment demonstrates both statistical and clinical significance")
    print(f"   - Response rate: {response_rate:.0f}% (exceeds 60% threshold)")
    print(f"   - Mean reduction: {mean_reduction:.1f} mmHg (clinically meaningful)")
elif response_rate >= 50:
    print("🟡 CONDITIONAL: Consider approval with post-market surveillance")
else:
    print("❌ REJECT: Insufficient evidence of efficacy")

# Number Needed to Treat (NNT)
nnt = 1 / (response_rate / 100)
print(f"\n💊 Number Needed to Treat (NNT): {nnt:.1f}")
print(f"(We need to treat ~{int(np.ceil(nnt))} patients for 1 to benefit)")
