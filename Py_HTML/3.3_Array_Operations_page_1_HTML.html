📝 Code Example: Glucose Monitoring Made Easy

Patient glucose readings (mg/dL) over 5 days 
import numpy as np

# Patient glucose readings (mg/dL) over 5 days
# Think of this as your patient's digital health diary! 📔
patient_glucose = np.array([95, 110, 105, 98, 102])

🔍 Quick Analysis - Finding the Average

Calculate average glucose level 
# Calculate average glucose level
# No loops needed - NumPy does the heavy lifting! 💪
avg_glucose = np.mean(patient_glucose)
print(f"Average glucose: {avg_glucose:.1f} mg/dL")
print(f"Verdict: {'Normal ✅' if avg_glucose < 100 else 'Keep monitoring 👀'}")

Blood Pressure Analysis 
# Multiple patients' systolic blood pressure readings
bp_readings = np.array([120, 135, 142, 118, 125, 139])

# Convert mmHg to other units for international standards
bp_kpa = bp_readings * 0.133322  # Convert to kPa
bp_torr = bp_readings * 1.0      # Already in Torr (same as mmHg)

print("Blood Pressure Conversions:")
print(f"mmHg: {bp_readings}")
print(f"kPa:  {bp_kpa.round(2)}")
print(f"Torr: {bp_torr}")

# Calculate pressure categories
# Normal: <120, Elevated: 120-129, Stage 1: 130-139, Stage 2: ≥140
normal_count = np.sum(bp_readings < 120)
elevated_count = np.sum((bp_readings >= 120) & (bp_readings < 130))
stage1_count = np.sum((bp_readings >= 130) & (bp_readings < 140))
stage2_count = np.sum(bp_readings >= 140)

print(f"\nPressure Categories:")
print(f"Normal (<120): {normal_count} patients")
print(f"Elevated (120-129): {elevated_count} patients")
print(f"Stage 1 (130-139): {stage1_count} patients")
print(f"Stage 2 (≥140): {stage2_count} patients")

💡 Complete Solution 
# Patient data: Heights (cm) and Weights (kg)
heights_cm = np.array([165, 180, 175, 160, 170, 185, 168, 172])
weights_kg = np.array([65, 80, 75, 55, 70, 90, 62, 68])

# Step 1: Convert heights to meters
heights_m = heights_cm / 100

# Step 2: Calculate BMI
bmi_values = weights_kg / (heights_m ** 2)

# Step 3: Categorize BMI values
underweight = bmi_values < 18.5
normal = (bmi_values >= 18.5) & (bmi_values < 25)
overweight = (bmi_values >= 25) & (bmi_values < 30)
obese = bmi_values >= 30

# Detailed analysis
print("🏥 Population BMI Analysis Report")
print("=" * 40)
print(f"Total patients analyzed: {len(bmi_values)}")
print(f"Average BMI: {np.mean(bmi_values):.1f}")
print(f"BMI Range: {np.min(bmi_values):.1f} - {np.max(bmi_values):.1f}")

print(f"\n📊 BMI Distribution:")
print(f"Underweight (<18.5):  {np.sum(underweight)} patients ({np.mean(underweight)*100:.1f}%)")
print(f"Normal (18.5-24.9):   {np.sum(normal)} patients ({np.mean(normal)*100:.1f}%)")
print(f"Overweight (25-29.9): {np.sum(overweight)} patients ({np.mean(overweight)*100:.1f}%)")
print(f"Obese (≥30):          {np.sum(obese)} patients ({np.mean(obese)*100:.1f}%)")

# Detailed patient breakdown
print(f"\n👥 Individual Patient Analysis:")
for i in range(len(bmi_values)):
    height_m = heights_m[i]
    weight = weights_kg[i]
    bmi = bmi_values[i]
    
    if bmi < 18.5:
        category = "Underweight"
        emoji = "⬇️"
    elif bmi < 25:
        category = "Normal"
        emoji = "✅"
    elif bmi < 30:
        category = "Overweight"
        emoji = "⚠️"
    else:
        category = "Obese"
        emoji = "🚨"
    
    print(f"Patient {i+1}: {height_m:.2f}m, {weight}kg → BMI {bmi:.1f} {emoji} {category}")

Advanced Mathematical Operations 
# Logarithmic transformations (common in biostatistics)
viral_loads = np.array([1000, 50000, 100000, 5000, 25000])

# Log transformation to normalize skewed data
log_viral_loads = np.log10(viral_loads)
print(f"Original viral loads: {viral_loads}")
print(f"Log10 transformed:    {log_viral_loads.round(2)}")

# Square root transformation for count data
white_cell_counts = np.array([4500, 8500, 6200, 7800, 5400])
sqrt_counts = np.sqrt(white_cell_counts)
print(f"\nWhite cell counts: {white_cell_counts}")
print(f"Square root transform: {sqrt_counts.round(1)}")

# Exponential functions for growth modeling
days = np.array([0, 1, 2, 3, 4, 5])

🎯 Trigonometric Functions in Medical Modeling

Circadian Rhythm Modeling 
# Model circadian body temperature variation
hours = np.arange(0, 24, 0.5)  # Every 30 minutes for 24 hours
baseline_temp = 98.6  # Normal body temperature in °F
amplitude = 1.8       # Temperature variation amplitude
phase_shift = 6       # Peak occurs at 6 PM (18:00)

# Circadian temperature model
# Temperature peaks in evening, lowest in early morning
temperature = baseline_temp + amplitude * np.sin(2 * np.pi * (hours - phase_shift) / 24)

print("🌡️ Circadian Body Temperature Model")
print("=" * 40)
print(f"Time (hrs)  Temperature (°F)")
print("-" * 25)

# Show temperature every 4 hours
for i in range(0, len(hours), 8):  # Every 4 hours
    hour = hours[i]
    temp = temperature[i]
    time_str = f"{int(hour):02d}:{int((hour % 1) * 60):02d}"
    print(f"{time_str}       {temp:.1f}°F")

# Find temperature extremes
min_temp_idx = np.argmin(temperature)
max_temp_idx = np.argmax(temperature)
print(f"\nDaily Temperature Range:")
print(f"Minimum: {temperature[min_temp_idx]:.1f}°F at {hours[min_temp_idx]:.1f} hours")
print(f"Maximum: {temperature[max_temp_idx]:.1f}°F at {hours[max_temp_idx]:.1f} hours")
