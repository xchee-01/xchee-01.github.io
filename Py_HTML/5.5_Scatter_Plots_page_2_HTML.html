🔬 Statistical Enhancement: Adding Trend Lines and Correlations

Medical researchers don't just want to see data points – they need to understand relationships! Let's add statistical insights to our scatter plots.

📈 Linear Regression and Correlation Coefficients

# Generate data for hypertension study
np.random.seed(200)
n_patients = 120

# Age of patients
age = np.random.uniform(30, 80, n_patients)

# Blood pressure increases with age (with variation)
systolic_bp = 100 + 0.8 * age + np.random.normal(0, 12, n_patients)
# Calculate correlation coefficient
from scipy import stats
correlation, p_value = stats.pearsonr(age, systolic_bp)
# Perform linear regression
slope, intercept = np.polyfit(age, systolic_bp, 1)
regression_line = slope * age + intercept
Linear Regression Basics

What does the slope value in a linear regression equation represent in the context of age vs blood pressure?

# Create enhanced statistical plot
plt.figure(figsize=(12, 8))
# Plot scatter points
plt.scatter(age, systolic_bp, alpha=0.6, s=60, c='navy', edgecolor='black', linewidth=0.5)
# Add regression line
plt.plot(age, regression_line, 'r-', linewidth=2, label=f'Linear fit: BP = {slope:.1f} × Age + {intercept:.1f}')
# Add confidence interval (95%)
from scipy import stats
predict_error = systolic_bp - regression_line
ci = 1.96 * np.std(predict_error)  # 95% confidence interval

plt.fill_between(age, regression_line - ci, regression_line + ci, 
                 alpha=0.2, color='red', label='95% Confidence Interval')
Confidence Intervals

What does a 95% confidence interval around a regression line indicate?

# Add correlation info to plot
textstr = f'Pearson r = {correlation:.3f}\np-value = {p_value:.4f}'
plt.text(0.05, 0.95, textstr, transform=plt.gca().transAxes,
         fontsize=12, verticalalignment='top',
         bbox=dict(boxstyle='round', facecolor='wheat', alpha=0.8))
# Finalize plot with labels
plt.title('Age-Related Blood Pressure Changes in Hypertension Study', fontsize=16, pad=20)
plt.xlabel('Patient Age (years)', fontsize=14)
plt.ylabel('Systolic Blood Pressure (mmHg)', fontsize=14)
plt.legend(loc='lower right')
plt.grid(True, alpha=0.3)
plt.show()
Statistical Insight: A p-value < 0.05 indicates a statistically significant relationship. In medical research, we also consider clinical significance – even small correlations can be meaningful for patient care! 📊

🎯 Identifying and Highlighting Outliers

In medical data, outliers might represent:

  • Patients with rare genetic variants
  • Measurement errors needing investigation
  • Candidates for personalized treatment protocols
# Generate clinical trial data with outliers
np.random.seed(300)
n_patients = 150

# Most patients respond normally to treatment
dose = np.random.uniform(10, 100, n_patients)
response = 20 + 0.5 * dose + np.random.normal(0, 8, n_patients)

# Add some outliers (unusual responders)
outlier_indices = [10, 25, 67, 89, 134]
for idx in outlier_indices:
    if idx < len(response):
        # Some are hyper-responders, some are non-responders
        if idx % 2 == 0:
            response[idx] = response[idx] + 40  # Hyper-responder
        else:
            response[idx] = 10  # Non-responder
# Identify outliers using z-score method
z_scores = np.abs(stats.zscore(response))
outlier_mask = z_scores > 2.5  # Points beyond 2.5 standard deviations
Outlier Detection

What does a z-score of 2.5 indicate about a data point?

# Create plot with highlighted outliers
plt.figure(figsize=(12, 8))
# Plot normal responders
plt.scatter(dose[~outlier_mask], response[~outlier_mask], 
           alpha=0.6, s=60, c='blue', label='Normal Responders')
# Highlight outliers with different color and size
plt.scatter(dose[outlier_mask], response[outlier_mask], 
           alpha=0.8, s=200, c='red', marker='*', 
           edgecolor='darkred', linewidth=2, label='Unusual Responders')
# Add annotations for outliers
for i, (d, r) in enumerate(zip(dose[outlier_mask], response[outlier_mask])):
    plt.annotate(f'Patient {np.where(outlier_mask)[0][i]}',
                xy=(d, r), xytext=(d+5, r+5),
                arrowprops=dict(arrowstyle='->', color='red', alpha=0.7),
                fontsize=10, color='darkred')
# Complete the clinical plot
plt.title('Drug Response Variability: Identifying Unusual Responders', fontsize=16, pad=20)
plt.xlabel('Drug Dose (mg)', fontsize=14)
plt.ylabel('Clinical Response Score', fontsize=14)
plt.legend()
plt.grid(True, alpha=0.3)
plt.show()
Precision Medicine Note: These "outliers" might have genetic variants affecting drug metabolism. In precision health, today's outlier could be tomorrow's personalized treatment group! 🧬

📐 Advanced Customization Techniques

# Generate comprehensive patient cohort data
np.random.seed(500)
n_patients = 300

# Diabetes study: HbA1c vs Fasting Glucose with complications
hba1c = np.random.normal(7.5, 1.5, n_patients)
hba1c = np.clip(hba1c, 5.0, 12.0)

# Fasting glucose correlates with HbA1c
fasting_glucose = 20 * hba1c + np.random.normal(0, 15, n_patients)
fasting_glucose = np.clip(fasting_glucose, 70, 300)

# Complication status (based on control levels)
complications = (hba1c > 8.5) | (fasting_glucose > 180)
Diabetes Markers

In diabetes management, what HbA1c level typically indicates poor glycemic control requiring intervention?

# Create publication-quality figure with custom style
plt.style.use('seaborn-v0_8-paper')  # Clean academic style
fig, ax = plt.subplots(figsize=(10, 8), dpi=300)  # High DPI for publications
# Create scatter plot with custom markers
# Plot patients without complications
ax.scatter(hba1c[~complications], fasting_glucose[~complications],
          s=40, c='forestgreen', marker='o', alpha=0.7,
          edgecolor='darkgreen', linewidth=0.5,
          label='Well-controlled (n={})'.format(np.sum(~complications)))

# Plot patients with complications
ax.scatter(hba1c[complications], fasting_glucose[complications],
          s=60, c='crimson', marker='^', alpha=0.7,
          edgecolor='darkred', linewidth=0.8,
          label='With complications (n={})'.format(np.sum(complications)))

💊 Personalized Medicine: Pharmacogenomics

Scatter plots help identify how genetic variations affect drug response:

# Simulate pharmacogenomics study
np.random.seed(600)

# Three genetic variants with different drug clearance rates
n_per_variant = 60

# Wild type (normal)
drug_dose_wt = np.random.uniform(50, 200, n_per_variant)
blood_level_wt = drug_dose_wt * np.random.normal(1.0, 0.15, n_per_variant)

# Heterozygous (reduced clearance)
drug_dose_het = np.random.uniform(50, 200, n_per_variant)
blood_level_het = drug_dose_het * np.random.normal(1.5, 0.2, n_per_variant)

# Homozygous (poor clearance)
drug_dose_hom = np.random.uniform(50, 200, n_per_variant)
blood_level_hom = drug_dose_hom * np.random.normal(2.2, 0.25, n_per_variant)
Drug Metabolism

A patient with homozygous variant for drug-metabolizing enzyme shows blood levels 2.2x higher than wild type. What dosing adjustment is most appropriate?

# Create personalized medicine visualization
fig, (ax1, ax2) = plt.subplots(1, 2, figsize=(16, 6))

# Left panel: Raw data
ax1.scatter(drug_dose_wt, blood_level_wt, c='green', alpha=0.6, s=50, label='Wild Type')
ax1.scatter(drug_dose_het, blood_level_het, c='orange', alpha=0.6, s=50, label='Heterozygous')
ax1.scatter(drug_dose_hom, blood_level_hom, c='red', alpha=0.6, s=50, label='Homozygous Variant')

# Add therapeutic window
ax1.axhspan(100, 200, alpha=0.2, color='lightblue', label='Therapeutic Range')
ax1.axhspan(200, 400, alpha=0.2, color='salmon', label='Toxic Range')

ax1.set_xlabel('Drug Dose (mg)', fontsize=12)
ax1.set_ylabel('Blood Level (ng/mL)', fontsize=12)
ax1.set_title('Pharmacogenomic Drug Response by Genotype', fontsize=14)
ax1.legend()
ax1.grid(True, alpha=0.3)

# Right panel: Dosing recommendations
genotypes = ['Wild Type', 'Heterozygous', 'Homozygous']
recommended_doses = [150, 100, 68]
dose_ranges = [25, 20, 15]

x_pos = np.arange(len(genotypes))
ax2.bar(x_pos, recommended_doses, yerr=dose_ranges, 
        color=['green', 'orange', 'red'], alpha=0.7,
        error_kw={'linewidth': 2, 'capsize': 5})

ax2.set_ylabel('Recommended Dose (mg)', fontsize=12)
ax2.set_title('Personalized Dosing Guidelines', fontsize=14)
ax2.set_xticks(x_pos)
ax2.set_xticklabels(genotypes)
ax2.grid(True, axis='y', alpha=0.3)

plt.tight_layout()
plt.show()
