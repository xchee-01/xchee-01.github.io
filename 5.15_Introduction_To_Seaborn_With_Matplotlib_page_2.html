# Create Multi-Dimensional Patient Risk Analysis # ============================================== # ============================================== # ============================================== # Generate comprehensive patient data with multiple risk factors # ================================================== np.random.seed(42) n_patients = 300 # Create correlated health metrics (realistic relationships) # ================================================== age = np.random.normal(55, 15, n_patients) bmi = 20 + 0.1 * age + np.random.normal(0, 4, n_patients) hdl = 70 - 0.3 * bmi + np.random.normal(0, 10, n_patients) # Good cholesterol ldl = 50 + 0.5 * age + 2 * bmi + np.random.normal(0, 20, n_patients) # Bad cholesterol glucose = 70 + 0.2 * age + 1.5 * bmi + np.random.normal(0, 15, n_patients) # Create risk categories based on multiple factors # ================================================ risk_score = ( (ldl > 160) * 1 + (hdl < 40) * 1 + (glucose > 100) * 1 + (bmi > 30) * 1 ) risk_category = pd.cut(risk_score, bins=[-1, 0, 1, 2, 4], labels=['Healthy', 'Low Risk', 'Moderate Risk', 'High Risk']) # Combine into DataFrame # ====================== risk_analysis = pd.DataFrame({ 'age': age, 'bmi': bmi, 'hdl_cholesterol': hdl, 'ldl_cholesterol': ldl, 'fasting_glucose': glucose, 'risk_category': risk_category }) # Create a professional pair plot # =============================== g = sns.pairplot(risk_analysis, hue='risk_category', diag_kind='kde', plot_kws={'alpha': 0.6}, height=2.5, palette={'Healthy': '#2ecc71', # Green 'Low Risk': '#3498db', # Blue 'Moderate Risk': '#f39c12', # Orange 'High Risk': '#e74c3c'}) # Red # Add overall title # ================= g.fig.suptitle('Multi-Dimensional Patient Risk Factor Analysis\n' + 'Cardiovascular Disease Risk Assessment (n=300)', y=1.02, fontsize=16, fontweight='bold') # Adjust layout # ============= plt.tight_layout() plt.show()

# Technique 1: Heatmaps for Biomarker Correlation Analysis # ================================================== # ======================================================== # ================================================== # Calculate correlation matrix # ============================ biomarkers = risk_analysis[['age', 'bmi', 'hdl_cholesterol', 'ldl_cholesterol', 'fasting_glucose']] correlation_matrix = biomarkers.corr() # Create a mask for the upper triangle (avoid redundancy) # ================================================== mask = np.triu(np.ones_like(correlation_matrix, dtype=bool)) # Set up the figure # ================= plt.figure(figsize=(10, 8)) # Create the heatmap with custom styling # ====================================== sns.heatmap(correlation_matrix, mask=mask, annot=True, # Show correlation values fmt='.2f', # Two decimal places cmap='coolwarm', # Diverging color scheme vmin=-1, vmax=1, # Correlation ranges from -1 to 1 center=0, # Center colormap at 0 square=True, # Square cells linewidths=1, # Cell borders cbar_kws={"shrink": .8, "label": "Correlation Coefficient"}) plt.title('Biomarker Correlation Matrix\n' + 'Understanding Interconnected Risk Factors', fontsize=16, fontweight='bold', pad=20) # Rotate labels for readability # ============================= plt.xticks(rotation=45, ha='right') plt.yticks(rotation=0) plt.tight_layout() plt.show()

Learning Checkpoint
Why do we create a mask for the upper triangle in the correlation heatmap?
To make the heatmap more colorful
Aesthetic improvement is the goal.
Incorrect. The mask is not for aesthetic purposes but for data clarity.
To avoid showing redundant information since correlation matrices are symmetric
Eliminates duplicate information in symmetric matrices.
Correct! Correlation matrices are symmetric (correlation of A with B equals correlation of B with A). Masking the upper triangle removes duplicate information and makes the visualization cleaner and easier to read.
To hide negative correlations
Negative values are being concealed.
Incorrect. The mask doesn't specifically hide negative correlations; it hides redundant information regardless of sign.
To reduce computational complexity
Performance optimization is the primary reason.
Incorrect. The mask is applied after computation for visualization clarity, not computational efficiency.

💡 Clinical Insight: Strong correlations between biomarkers can indicate:

  • Shared biological pathways
  • Potential for multicollinearity in predictive models
  • Opportunities for simplified screening (measure one, infer others)
# Technique 2: Joint Plots for Detailed Relationship Analysis # ================================================== # =========================================================== # ================================================== # Create a joint plot examining the BMI-Glucose relationship # ================================================== g = sns.jointplot(data=risk_analysis, x='bmi', y='fasting_glucose', hue='risk_category', kind='scatter', height=8, palette={'Healthy': '#2ecc71', 'Low Risk': '#3498db', 'Moderate Risk': '#f39c12', 'High Risk': '#e74c3c'}, marginal_kws={'bins': 20, 'alpha': 0.6}) # Add reference lines for clinical thresholds # =========================================== g.ax_joint.axvline(x=25, color='gray', linestyle='--', alpha=0.5) # Overweight g.ax_joint.axvline(x=30, color='gray', linestyle='--', alpha=0.5) # Obese g.ax_joint.axhline(y=100, color='gray', linestyle='--', alpha=0.5) # Prediabetes g.ax_joint.axhline(y=126, color='gray', linestyle='--', alpha=0.5) # Diabetes # Add labels for reference lines # ============================== g.ax_joint.text(25.5, 145, 'Overweight', rotation=90, va='bottom', ha='left', fontsize=9, alpha=0.7) g.ax_joint.text(30.5, 145, 'Obese', rotation=90, va='bottom', ha='left', fontsize=9, alpha=0.7) g.ax_joint.text(17, 101, 'Prediabetes', fontsize=9, alpha=0.7) g.ax_joint.text(17, 127, 'Diabetes', fontsize=9, alpha=0.7) # Set labels # ========== g.set_axis_labels('Body Mass Index (BMI)', 'Fasting Glucose (mg/dL)', fontsize=12, fontweight='bold') # Add title # ========= g.fig.suptitle('BMI vs Fasting Glucose: Identifying Metabolic Risk Patterns', fontsize=14, fontweight='bold', y=1.02) plt.tight_layout() plt.show()
