GridSpec Understanding
What does `gs[0, :2]` mean in the GridSpec syntax?
First row, first two columns
A) First row, first two columns
Correct! The syntax [0, :2] means row 0 (first row) and columns from start up to (but not including) 2, so columns 0 and 1.
First column, first two rows
B) First column, first two rows
Not quite. The syntax is [row, column], not [column, row]. Also, :2 refers to columns, not rows.
All rows, first two columns
C) All rows, first two columns
That would be written as gs[:, :2]. The 0 specifically refers to the first row only.
First two elements of the grid
D) First two elements of the grid
GridSpec uses 2D indexing [row, column], not linear indexing. This syntax spans specific rows and columns.

# Generate realistic ECG-like data with cardiac cycles # ================================================== def generate_ecg_cycle(duration=0.8, sampling_rate=1000): """Generate one cardiac cycle (PQRST complex)""" t = np.linspace(0, duration, int(duration * sampling_rate)) ecg = np.zeros_like(t) # P wave (atrial depolarization) p_wave = 0.1 * np.exp(-((t - 0.1)**2) / 0.01) # QRS complex (ventricular depolarization) q_wave = -0.15 * np.exp(-((t - 0.19)**2) / 0.0001) r_wave = 1.0 * np.exp(-((t - 0.2)**2) / 0.0001) s_wave = -0.15 * np.exp(-((t - 0.21)**2) / 0.0001) # T wave (ventricular repolarization) t_wave = 0.2 * np.exp(-((t - 0.4)**2) / 0.02) ecg = p_wave + q_wave + r_wave + s_wave + t_wave return t, ecg # Generate multiple cardiac cycles # ================================ time_ecg = [] ecg_signal = [] for i in range(5): # 5 heartbeats t, ecg = generate_ecg_cycle() time_ecg.extend(t + i * 0.8) ecg_signal.extend(ecg) time_ecg = np.array(time_ecg) ecg_signal = np.array(ecg_signal) # Add realistic noise # =================== ecg_signal += 0.05 * np.random.normal(0, 1, len(ecg_signal)) # Plot ECG trace with clinical annotations # ======================================== ax_ecg.plot(time_ecg, ecg_signal, 'g-', linewidth=1.5) ax_ecg.set_title('ECG Monitor - Lead II', fontsize=14, fontweight='bold', color='darkgreen') ax_ecg.set_xlabel('Time (seconds)') ax_ecg.set_ylabel('Amplitude (mV)') ax_ecg.grid(True, alpha=0.3, which='both', linestyle=':') ax_ecg.set_ylim(-0.5, 1.5) # Add clinical annotations # ======================== ax_ecg.annotate('R', xy=(0.2, 1.0), xytext=(0.2, 1.3), arrowprops=dict(arrowstyle='->', color='red'), fontsize=10, ha='center', color='red') ax_ecg.text(3.5, 1.3, 'Normal Sinus Rhythm', fontsize=12, bbox=dict(boxstyle='round', facecolor='lightgreen')) # Add background grid like real ECG paper # ======================================= ax_ecg.set_axisbelow(True) ax_ecg.grid(True, which='major', alpha=0.4, color='red', linewidth=0.5) ax_ecg.grid(True, which='minor', alpha=0.2, color='red', linewidth=0.25) ax_ecg.minorticks_on()
