Your Mission: Create a Python script that performs a complete system check and generates a professional lab report.

Create a new file called medical_environment_diagnostic.py:

# 🔬 Precision Health Programming Environment Diagnostic
import sys
from datetime import datetime

print("🏥 PRECISION HEALTH LAB DIAGNOSTIC REPORT")
print("=" * 50)
print(f"📅 Diagnostic Date: {datetime.now().strftime('%Y-%m-%d %H:%M:%S')}")
print(f"🖥️  System Platform: {sys.platform}")
print()

# Test 1: Python Core System
print("🧬 CORE SYSTEM ANALYSIS")
print("-" * 25)
print(f"✅ Python Version: {sys.version.split()[0]}")
print(f"✅ Python Executable: {sys.executable}")

# Test 2: Medical Research Libraries
print("\n🔬 MEDICAL RESEARCH TOOLKIT VERIFICATION")
print("-" * 40)

required_libraries = [
    ('numpy', 'NumPy - Numerical computing for medical data'),
    ('pandas', 'Pandas - Medical data analysis and manipulation'),
    ('matplotlib', 'Matplotlib - Medical data visualization'),
    ('jupyter', 'Jupyter - Interactive research notebooks')
]

for lib_name, description in required_libraries:
    try:
        lib = __import__(lib_name)
        version = getattr(lib, '__version__', 'Version unknown')
        print(f"✅ {description}: v{version}")
    except ImportError:
        print(f"❌ {description}: NOT INSTALLED")
        print(f"   Install with: pip install {lib_name}")

# Test 3: Performance Benchmark for Medical Data Processing
print("\n⚡ PERFORMANCE BENCHMARK")
print("-" * 23)

import time
import numpy as np

# Simulate processing 1 million patient data points
start_time = time.time()
medical_data = np.random.normal(98.6, 1.5, 1000000)  # Body temperatures
average_temp = np.mean(medical_data)
processing_time = time.time() - start_time

print(f"✅ Processed 1M patient records in {processing_time:.4f} seconds")
print(f"✅ Average patient temperature: {average_temp:.2f}°F")
print()
print("🎉 DIAGNOSTIC COMPLETE: Your environment is ready for precision health programming!")
