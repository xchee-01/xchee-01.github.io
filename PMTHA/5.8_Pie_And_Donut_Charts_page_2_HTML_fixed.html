Cancer Staging Analysis: Revealing Hidden Patterns

# Simulating tumor size distributions across cancer stages np.random.seed(789) # Stage I: Mostly small, but some rapid growers stage_1_tumors = np.concatenate([ np.random.normal(2, 0.5, 80), # Main population np.random.normal(1, 0.3, 20) # Very early detection ]) # Stage II: Starting to show heterogeneity stage_2_tumors = np.concatenate([ np.random.normal(4, 0.8, 70), # Typical progression np.random.normal(5.5, 0.5, 30) # Aggressive subtype ]) # Stage III: Complex, multimodal distribution stage_3_tumors = np.concatenate([ np.random.normal(7, 1.2, 50), # Standard progression np.random.normal(9, 1, 30), # Highly aggressive np.random.normal(5, 0.8, 20) # Slow-growing variant ])
Copied!

🧬 Biology note: These multimodal distributions often reflect different tumor subtypes with distinct genetic profiles – crucial for targeted therapy selection!

# Creating an informative violin plot plt.figure(figsize=(10, 6)) # Create violin plot with all the bells and whistles parts = plt.violinplot([stage_1_tumors, stage_2_tumors, stage_3_tumors], positions=[1, 2, 3], showmeans=True, # Show mean as a line showmedians=True, # Show median as a line showextrema=True) # Show min/max # Color progression: green (early) to red (advanced) colors = ['#90EE90', '#FFD700', '#FF6B6B'] for pc, color in zip(parts['bodies'], colors): pc.set_facecolor(color) pc.set_alpha(0.7) # Clinical context plt.xlabel('Cancer Stage') plt.ylabel('Tumor Size (cm)') plt.title('Tumor Size Distribution by Cancer Stage') plt.xticks([1, 2, 3], ['Stage I', 'Stage II', 'Stage III']) plt.grid(True, axis='y', alpha=0.3) # Surgical decision threshold plt.axhline(y=5, color='red', linestyle='--', label='Surgical Recommendation Threshold (5cm)') plt.legend() plt.show()
Copied!

💡 Clinical Insights from the Violin Shapes:

  • Stage I: Tight distribution with small secondary peak (early detection success!)
  • Stage II: Widening distribution (heterogeneous growth patterns emerging)
  • Stage III: Multiple peaks (different tumor subtypes requiring personalized treatment)

Impact on Treatment: Oncologists use these patterns to identify patients who might benefit from neoadjuvant therapy (shrinking tumors before surgery) versus immediate surgical intervention.

A violin plot showing: - Healthy controls with low CRP values (mostly below 3 mg/L) - Rheumatoid arthritis with consistently elevated CRP (centered around 12 mg/L) - Lupus with bimodal distribution showing two distinct peaks - Clinical reference lines at 3 mg/L and 10 mg/L

Comprehensive Liver Function Dashboard

Let's create a clinical dashboard that hepatologists dream about – simultaneous visualization of multiple liver enzymes across disease states!

# Liver enzyme simulation across disease progression np.random.seed(567) # Disease conditions we're comparing conditions = ['Healthy', 'Fatty Liver', 'Hepatitis', 'Cirrhosis'] # ALT (Alanine Aminotransferase) - inflammation marker alt_healthy = np.random.normal(25, 8, 100) alt_fatty = np.random.lognormal(3.7, 0.4, 100) # Mild elevation alt_hepatitis = np.random.lognormal(4.5, 0.5, 100) # Significant elevation alt_cirrhosis = np.random.lognormal(4.2, 0.6, 100) # Variable # AST (Aspartate Aminotransferase) - cell damage marker ast_healthy = np.random.normal(22, 6, 100) ast_fatty = np.random.normal(35, 10, 100) ast_hepatitis = np.random.lognormal(4.3, 0.5, 100) ast_cirrhosis = np.random.lognormal(4.6, 0.6, 100) # Highest
Copied!

💡 Clinical note: The AST/ALT ratio helps differentiate liver diseases. Alcoholic liver disease typically shows AST > ALT, while viral hepatitis shows ALT > AST!

# Create comprehensive comparison dashboard fig, axes = plt.subplots(2, 2, figsize=(14, 10)) # ALT box plot (top left) ax1 = axes[0, 0] bp1 = ax1.boxplot([alt_healthy, alt_fatty, alt_hepatitis, alt_cirrhosis], labels=conditions, patch_artist=True) for patch in bp1['boxes']: patch.set_facecolor('lightblue') ax1.set_ylabel('ALT (U/L)') ax1.set_title('ALT Levels - Box Plot') ax1.axhline(y=40, color='red', linestyle='--', alpha=0.5) ax1.grid(True, alpha=0.3) # ALT violin plot (top right) ax2 = axes[0, 1] vp1 = ax2.violinplot([alt_healthy, alt_fatty, alt_hepatitis, alt_cirrhosis], positions=[1, 2, 3, 4], showmeans=True, showmedians=True) for pc in vp1['bodies']: pc.set_facecolor('lightblue') pc.set_alpha(0.7) ax2.set_xticks([1, 2, 3, 4]) ax2.set_xticklabels(conditions) ax2.set_ylabel('ALT (U/L)') ax2.set_title('ALT Levels - Violin Plot') ax2.axhline(y=40, color='red', linestyle='--', alpha=0.5) ax2.grid(True, alpha=0.3) # AST box plot (bottom left) ax3 = axes[1, 0] bp2 = ax3.boxplot([ast_healthy, ast_fatty, ast_hepatitis, ast_cirrhosis], labels=conditions, patch_artist=True) for patch in bp2['boxes']: patch.set_facecolor('lightcoral') ax3.set_ylabel('AST (U/L)') ax3.set_title('AST Levels - Box Plot') ax3.axhline(y=40, color='red', linestyle='--', alpha=0.5) ax3.grid(True, alpha=0.3) # AST violin plot (bottom right) ax4 = axes[1, 1] vp2 = ax4.violinplot([ast_healthy, ast_fatty, ast_hepatitis, ast_cirrhosis], positions=[1, 2, 3, 4], showmeans=True, showmedians=True) for pc in vp2['bodies']: pc.set_facecolor('lightcoral') pc.set_alpha(0.7) ax4.set_xticks([1, 2, 3, 4]) ax4.set_xticklabels(conditions) ax4.set_ylabel('AST (U/L)') ax4.set_title('AST Levels - Violin Plot') ax4.axhline(y=40, color='red', linestyle='--', alpha=0.5) ax4.grid(True, alpha=0.3) plt.suptitle('Liver Function Test Comparison Across Disease States', fontsize=16) plt.tight_layout() plt.show()
Copied!

📊 Dashboard Insights:

  • Box plots: Quick quartile comparison and outlier detection
  • Violin plots: Reveal skewed distributions in disease states
  • Together: Complete picture for clinical decision-making

Clinical Application: This dashboard format is standard in electronic health records (EHRs) for tracking disease progression and treatment response over time.

🔬 Advanced Topic: Statistical Annotations for Evidence-Based Medicine

In the world of clinical research, pretty pictures aren't enough – we need statistical evidence! Let's add p-values and confidence intervals to transform visualizations into publication-ready figures.

Treatment Response Analysis with Statistical Rigor

from scipy import stats # Simulating inflammatory biomarker response to treatment np.random.seed(678) # Treatment timeline data baseline_inflammation = np.random.lognormal(3.5, 0.5, 50) week_4_inflammation = np.random.lognormal(3.0, 0.4, 50) week_12_inflammation = np.random.lognormal(2.5, 0.35, 50) # Statistical testing (Wilcoxon signed-rank for paired data) p_val_4wk = stats.wilcoxon(baseline_inflammation, week_4_inflammation).pvalue p_val_12wk = stats.wilcoxon(baseline_inflammation, week_12_inflammation).pvalue print(f"4-week p-value: {p_val_4wk:.4f}") print(f"12-week p-value: {p_val_12wk:.4f}")
Copied!
# Create publication-quality figure with annotations fig, ax = plt.subplots(figsize=(10, 8)) # Violin plots showing distribution changes positions = [1, 2, 3] vp = ax.violinplot([baseline_inflammation, week_4_inflammation, week_12_inflammation], positions=positions, showmeans=True, showmedians=True) # Color gradient showing improvement colors = ['#FF6B6B', '#FFE66D', '#4ECDC4'] # Red → Yellow → Teal for pc, color in zip(vp['bodies'], colors): pc.set_facecolor(color) pc.set_alpha(0.7) pc.set_edgecolor('black') pc.set_linewidth(1.5) # Overlay box plots for quartile information bp = ax.boxplot([baseline_inflammation, week_4_inflammation, week_12_inflammation], positions=positions, widths=0.2, patch_artist=False, showfliers=False, zorder=10)
Copied!
# Statistical annotation function def add_stat_annotation(ax, x1, x2, y, p_val, h=0.05): """Add significance bars with p-values""" # Draw the bracket ax.plot([x1, x1, x2, x2], [y, y+h, y+h, y], 'k-', linewidth=1.5) # Determine significance level if p_val < 0.001: sig_text = '***' elif p_val < 0.01: sig_text = '**' elif p_val < 0.05: sig_text = '*' else: sig_text = 'ns' # Add text ax.text((x1+x2)/2, y+h, sig_text, ha='center', va='bottom', fontsize=12) ax.text((x1+x2)/2, y+h+2, f'p={p_val:.3f}', ha='center', va='bottom', fontsize=9) # Add statistical comparisons y_max = max([baseline_inflammation.max(), week_4_inflammation.max(), week_12_inflammation.max()]) add_stat_annotation(ax, 1, 2, y_max*1.1, p_val_4wk) add_stat_annotation(ax, 1, 3, y_max*1.2, p_val_12wk) # Professional formatting ax.set_xticks(positions) ax.set_xticklabels(['Baseline', 'Week 4', 'Week 12']) ax.set_ylabel('Inflammatory Biomarker (pg/mL)', fontsize=12) ax.set_title('Treatment Response: Inflammatory Biomarker Reduction', fontsize=14) ax.axhline(y=20, color='green', linestyle='--', alpha=0.5, label='Target Level') ax.grid(True, axis='y', alpha=0.3) ax.legend() # Add response rate annotation responders_12wk = sum(week_12_inflammation < 20) / len(week_12_inflammation) * 100 ax.text(3.5, y_max*0.8, f'Response Rate:\n{responders_12wk:.1f}%', bbox=dict(boxstyle="round,pad=0.3", facecolor="lightgreen", alpha=0.5), fontsize=10, ha='center') plt.tight_layout() plt.show()
