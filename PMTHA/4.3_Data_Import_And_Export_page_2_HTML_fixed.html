🔬 Chapter 7: Error Handling - Because Real Data is Messy!

The Reality of Medical Data 🏥

In the perfect world of tutorials, data files always exist, formats are always correct, and encodings never cause problems. In the real world of healthcare... well, let's just say you need armor!

🛡️ Defensive Programming Saves Lives (Literally!)

def safe_import_patient_data(filename, critical=True): """ Safely import patient data with comprehensive error handling Args: filename: Path to patient data file critical: If True, system should halt on errors """ try: # Try standard import first data = pd.read_csv(filename) print(f"✅ Successfully imported {len(data)} patient records") return data except FileNotFoundError: print(f"❌ ERROR: File '{filename}' not found!") if critical: print("🚨 CRITICAL: Cannot proceed without patient data!") # In real system: Send alert to IT team raise else: print("⚠️ WARNING: Proceeding with empty dataset") return pd.DataFrame() except pd.errors.EmptyDataError: print(f"❌ ERROR: File '{filename}' is empty!") return pd.DataFrame() except pd.errors.ParserError as e: print(f"❌ ERROR: Cannot parse file - {str(e)}") print("🔧 Attempting recovery with error_bad_lines=False...") try: data = pd.read_csv(filename, error_bad_lines=False) print(f"✅ Recovered {len(data)} valid records") return data except: print("❌ Recovery failed - file may be corrupted") raise

🌍 Handling International Medical Data

def import_international_patient_data(filename): """Import data that might contain various languages and encodings""" encodings = ['utf-8', 'latin-1', 'iso-8859-1', 'cp1252'] for encoding in encodings: try: data = pd.read_csv(filename, encoding=encoding) print(f"✅ Successfully read with {encoding} encoding") # Validate critical columns exist required_columns = ['patient_id', 'diagnosis', 'treatment'] missing = set(required_columns) - set(data.columns) if missing: raise ValueError(f"Missing required columns: {missing}") return data except (UnicodeDecodeError, UnicodeError): continue except Exception as e: print(f"❌ Failed with {encoding}: {str(e)}") continue raise ValueError(f"Could not read {filename} with any known encoding")
Copied!

🔬 Chapter 8: Performance Optimization - Handling Big Medical Data

When Data Gets HUGE 🦣

Genomic sequencing can produce files with billions of rows. Medical imaging metadata can span terabytes. Here's how to handle data that would make ordinary programs cry!

🚄 Chunking - Eating the Elephant One Bite at a Time

def process_large_genomic_file(filename, chunk_size=10000): """Process massive genomic data files in manageable chunks""" total_mutations = 0 pathogenic_count = 0 # Process file in chunks for chunk_num, chunk in enumerate(pd.read_csv(filename, chunksize=chunk_size)): # Show progress print(f"Processing chunk {chunk_num + 1} ({len(chunk)} variants)...", end='\r') # Analyze this chunk total_mutations += len(chunk) pathogenic = chunk[chunk['clinical_significance'] == 'Pathogenic'] pathogenic_count += len(pathogenic) # Save important findings immediately if len(pathogenic) > 0: pathogenic.to_csv( f'pathogenic_variants_chunk_{chunk_num}.csv', mode='a', # Append mode header=(chunk_num == 0), # Only write header for first chunk index=False ) print(f"\n✅ Analysis complete!") print(f"Total variants analyzed: {total_mutations:,}") print(f"Pathogenic variants found: {pathogenic_count:,}") print(f"Pathogenic rate: {pathogenic_count/total_mutations*100:.2f}%")

class ClinicalDataPipeline: """Production-ready pipeline for daily clinical data processing""" def __init__(self, config_file='pipeline_config.json'): with open(config_file, 'r') as f: self.config = json.load(f) self.log = [] def run_daily_pipeline(self, date=None): """Execute complete daily clinical data pipeline""" if date is None: date = pd.Timestamp.now().date() print(f"🏥 Starting clinical pipeline for {date}") try: # Step 1: Import multiple data sources lab_results = self._import_lab_results(date) vital_signs = self._import_vital_signs(date) medications = self._import_medications(date) # Step 2: Merge and validate patient_data = self._merge_patient_data(lab_results, vital_signs, medications) validation_report = self._validate_data_quality(patient_data) # Step 3: Run clinical algorithms risk_scores = self._calculate_risk_scores(patient_data) alerts = self._generate_clinical_alerts(risk_scores) # Step 4: Export results self._export_clinical_dashboard(patient_data, risk_scores, alerts) self._send_urgent_notifications(alerts) # Step 5: Archive and log self._archive_daily_data(date, patient_data) self._log_pipeline_success(date, len(patient_data)) print(f"✅ Pipeline complete! def create_fhir_observation(patient_id, test_name, value, unit, timestamp): """Create FHIR-compliant observation resource""" observation = { "resourceType": "Observation", "id": f"obs-{patient_id}-{timestamp.strftime('%Y%m%d%H%M%S')}", "status": "final", "code": { "coding": [{ "system": "http://loinc.org", "code": get_loinc_code(test_name), # Map to standard code "display": test_name }] }, "subject": { "reference": f"Patient/{patient_id}" }, "effectiveDateTime": timestamp.isoformat(), "valueQuantity": { "value": value, "unit": unit, "system": "http://unitsofmeasure.org", "code": unit } } return observation # Example: Converting lab results to FHIR format def export_labs_to_fhir(lab_dataframe, output_file='fhir_observations.json'): """Convert lab results DataFrame to FHIR Bundle""" bundle = { "resourceType": "Bundle", "type": "collection", "entry": [] } for _, lab in lab_dataframe.iterrows(): observation = create_fhir_observation( lab['patient_id'], lab['test_name'], lab['result_value'], lab['unit'], lab['collection_time'] ) bundle['entry'].append({"resource": observation}) with open(output_file, 'w') as f: json.dump(bundle, f, indent=2) print(f"✅ Exported {len(bundle['entry'])} observations in FHIR format")

def validate_patient_data(df): """Ensure imported data meets medical standards""" issues = [] # Check for required columns required = ['patient_id', 'date_of_birth', 'gender'] missing = set(required) - set(df.columns) if missing: issues.append(f"Missing columns: {missing}") # Check for duplicates duplicates = df.duplicated(subset=['patient_id']).sum() if duplicates > 0: issues.append(f"Found {duplicates} duplicate patient IDs") # Validate data types and ranges if 'age' in df.columns: invalid_ages = ((df['age'] < 0) | (df['age'] > 150)).sum() if invalid_ages > 0: issues.append(f"Found {invalid_ages} invalid ages") return issues

Pitfall 1: Timezone Confusion
Which approach is better for handling timestamps in medical data?
Ignore timezones completely
timestamps = pd.to_datetime(df['admission_time'])
This can lead to confusion when comparing data from different time zones or during daylight saving transitions.
Always specify timezone for medical data
timestamps = pd.to_datetime(df['admission_time']).dt.tz_localize('UTC')
Correct! Always specifying timezone prevents confusion and ensures accurate temporal analysis across different locations and time periods.
Use local time only
Assume all data uses the local system timezone.
This assumption can be dangerous when dealing with multi-site studies or when systems change locations.
Convert everything to strings
Store timestamps as text to avoid timezone issues.
This prevents proper temporal analysis and sorting. Better to handle timezones properly than avoid them.
Pitfall 2: Excel Date Corruption
Excel might convert gene names to dates! (MARCH1 becomes Mar-1, SEPT9 becomes Sep-9). What's the solution?
Avoid Excel completely
Never use Excel for any genomic data.
While this avoids the problem, Excel is widely used in research. Better to know how to handle it properly.
Force text format for gene data
gene_data.to_excel('genes.xlsx', engine='xlsxwriter')
Correct! Using appropriate engines and formatting options can prevent Excel from automatically converting gene names to dates.
Rename all problematic genes
Change gene names to avoid date-like patterns.

Use CSV format only
Always export to CSV to avoid Excel formatting.
