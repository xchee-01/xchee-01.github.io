Time-Series Vital Signs

# Patient vital signs throughout the day vital_signs_log = { "06:00": {"temp": 98.2, "pulse": 68, "bp": "118/76", "o2_sat": 98}, "10:00": {"temp": 98.6, "pulse": 72, "bp": "120/80", "o2_sat": 97}, "14:00": {"temp": 99.1, "pulse": 78, "bp": "125/82", "o2_sat": 97}, "18:00": {"temp": 99.8, "pulse": 84, "bp": "130/85", "o2_sat": 96}, "22:00": {"temp": 100.2, "pulse": 88, "bp": "132/86", "o2_sat": 95} } # Analyze temperature trend print("🌡️ Temperature Trend Analysis:") temps = [] for time, vitals in vital_signs_log.items(): temp = vitals["temp"] temps.append(temp) status = "⚠️ FEVER" if temp >= 100.4 else "📈 Elevated" if temp > 99 else "✅ Normal" print(f" {time}: {temp}°F {status}") # Calculate statistics avg_temp = sum(temps) / len(temps) max_temp = max(temps) print(f"\nAverage: {avg_temp:.1f}°F") print(f"Peak: {max_temp}°F at ", end="") for time, vitals in vital_signs_log.items(): if vitals["temp"] == max_temp: print(time)

Finding Patterns and Alerts

# Check for concerning trends print("\n🚨 Clinical Alerts:") alerts_found = False for time, vitals in vital_signs_log.items(): alerts = [] if vitals["temp"] >= 100.4: alerts.append("Fever") if vitals["pulse"] > 100: alerts.append("Tachycardia") if vitals["o2_sat"] < 95: alerts.append("Low O2") if alerts: alerts_found = True print(f" {time}: {', '.join(alerts)}") if not alerts_found: print(" No critical alerts")

Given the current medications and interaction rules above, adding aspirin (antiplatelet) would result in:
No warnings
No interaction warnings

Bradycardia risk

Answer:

current_medications = { "warfarin": {"dose": "5mg", "class": "anticoagulant"}, "metoprolol": {"dose": "50mg", "class": "beta-blocker"}, "simvastatin": {"dose": "40mg", "class": "statin"} } interaction_rules = { ("anticoagulant", "antiplatelet"): "High bleeding risk", ("beta-blocker", "calcium-channel-blocker"): "Risk of bradycardia", ("statin", "fibrate"): "Increased myopathy risk" } new_drug = "aspirin" new_drug_class = "antiplatelet" print(f"💊 Checking interactions for {new_drug} ({new_drug_class})...") print("=" * 50) interactions_found = [] # Check each current medication for med_name, med_info in current_medications.items(): current_class = med_info["class"] # Check both directions of interaction if (current_class, new_drug_class) in interaction_rules: risk = interaction_rules[(current_class, new_drug_class)] interactions_found.append({ "drugs": f"{new_drug} + {med_name}", "risk": risk, "severity": "HIGH" if "High" in risk else "MODERATE" }) elif (new_drug_class, current_class) in interaction_rules: risk = interaction_rules[(new_drug_class, current_class)] interactions_found.append({ "drugs": f"{new_drug} + {med_name}", "risk": risk, "severity": "HIGH" if "High" in risk else "MODERATE" }) # Report findings if interactions_found: print("\n⚠️ DRUG INTERACTIONS DETECTED!") for interaction in interactions_found: severity_emoji = "🔴" if interaction["severity"] == "HIGH" else "🟡" print(f"\n{severity_emoji} {interaction['severity']} RISK") print(f" Drugs: {interaction['drugs']}") print(f" Risk: {interaction['risk']}") print(f" Action: Consult physician before prescribing") else: print("\n✅ No major interactions detected") print(f" {new_drug} appears safe with current medications")
Copied!
Learning Checkpoint
Why does the code check both `(current_class, new_drug_class)` and `(new_drug_class, current_class)` in the interaction_rules?
Performance
To make the code run faster
No, checking twice actually makes it slightly slower, not faster.
Bidirectional checking
Because drug interactions might be defined in either order in the rules dictionary
Correct! Drug interactions are bidirectional, but the interaction_rules dictionary might define them in either order. Checking both combinations ensures no interactions are missed regardless of how they're stored.
Duplicate checking
To check for duplicate medications
Incorrect. This checks for interactions between different drug classes, not duplicates.
Validation
To validate the drug classes exist
No, this is checking if the combination exists in the rules, not validating classes.

# Calculate treatment effectiveness experimental_improvements = [] control_improvements = [] for patient in clinical_trial: hba1c_change = patient["baseline_hba1c"] - patient["week_12_hba1c"] weight_change = patient["baseline_weight"] - patient["week_12_weight"] result = { "id": patient["patient_id"], "hba1c_improvement": hba1c_change, "weight_loss": weight_change } if patient["treatment_group"] == "experimental": experimental_improvements.append(hba1c_change) else: control_improvements.append(hba1c_change) # Calculate averages avg_experimental = sum(experimental_improvements) / len(experimental_improvements) avg_control = sum(control_improvements) / len(control_improvements) print("📊 Clinical Trial Results - Week 12") print("=" * 40) print(f"Experimental Group:") print(f" Average HbA1c reduction: {avg_experimental:.2f}%") print(f"Control Group:") print(f" Average HbA1c reduction: {avg_control:.2f}%") print(f"\n✨ Treatment Effect: {avg_experimental - avg_control:.2f}% additional reduction")

Building a Hospital Department System

# Hospital department database hospital_db = { "cardiology": { "patients": 45, "beds": 50, "head": "Dr. Sarah Kim", "specialties": ["interventional", "electrophysiology", "heart-failure"], "equipment": ["cath-lab", "echo", "stress-test"] }, "oncology": { "patients": 78, "beds": 80, "head": "Dr. Michael Chen", "specialties": ["medical", "radiation", "surgical"], "equipment": ["linear-accelerator", "pet-scanner", "infusion-center"] }, "neurology": { "patients": 32, "beds": 40, "head": "Dr. Jennifer Park", "specialties": ["stroke", "epilepsy", "movement-disorders"], "equipment": ["mri", "eeg", "emg"] } } # Find departments near capacity print("🏥 Department Capacity Analysis") print("=" * 40) for dept_name, dept_info in hospital_db.items(): occupancy = (dept_info["patients"] / dept_info["beds"]) * 100 status = "🔴 FULL" if occupancy >= 95 else "🟡 Near Capacity" if occupancy >= 80 else "🟢 Available" print(f"\n{dept_name.capitalize()}:") print(f" Chief: {dept_info['head']}") print(f" Occupancy: {occupancy:.0f}% ({dept_info['patients']}/{dept_info['beds']} beds) {status}") print(f" Specialties: {', '.join(dept_info['specialties'])}")

Learning Checkpoint
In the BMI by medication analysis, which line of code correctly adds a patient's BMI to the appropriate medication group?
Direct assignment
med_groups[med] = patient["bmi"]
This would replace the list with a single value, losing previous data.
Append to list
med_groups[med].append(patient["bmi"])
Correct! med_groups[med].append(patient["bmi"]) correctly appends the BMI value to the list associated with that medication type. The dictionary stores lists of BMI values for each medication.
Append dictionary
med_groups.append({med: patient["bmi"]})
med_groups is a dictionary, not a list. You can't append to it.
Reverse mapping
med_groups[patient["bmi"]] = med
This reverses the key-value relationship. We want medication as key, not BMI.

Learning Checkpoint
When accessing patient data that might not exist in a dictionary, which method is safest to use?
Direct access
patient_dict["allergies"]
This will raise a KeyError if "allergies" doesn't exist.
get() with default
patient_dict.get("allergies", [])
Correct! The get() method with a default value (empty list in this case) safely handles missing keys without raising errors, making it ideal for medical data where fields might be optional.
keys() method
patient_dict.keys("allergies")
keys() doesn't take arguments and returns all keys, not values.
find() method
patient_dict.find("allergies")
Dictionaries don't have a find() method. That's for strings.
