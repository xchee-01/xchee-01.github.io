# Simulating a randomized controlled trial for a new hypertension drug # ================================================== np.random.seed(123) days = np.arange(0, 90) # 3-month trial # Generate realistic treatment responses # ====================================== def generate_treatment_response(baseline, daily_reduction, noise_level): """Simulate treatment response with realistic variability""" # Add weekly cycles (patients often miss weekend doses) weekly_pattern = 2 * np.sin(2 * np.pi * days / 7) response = baseline - daily_reduction * days + weekly_pattern + \ np.random.normal(0, noise_level, len(days)) return response # Three treatment arms # ==================== placebo = generate_treatment_response(160, 0.05, 8) drug_low_dose = generate_treatment_response(160, 0.15, 6) drug_high_dose = generate_treatment_response(160, 0.25, 6) # Create figure with shared x-axis for easy comparison # ================================================== fig, (ax1, ax2, ax3) = plt.subplots(3, 1, figsize=(12, 10), sharex=True, sharey=True) fig.patch.set_facecolor('white') # Common styling function # ======================= def style_treatment_plot(ax, data, color, label, group_name): """Apply consistent styling to treatment plots""" # Plot with confidence interval ax.plot(days, data, color=color, linewidth=2, label=label) # Add rolling mean for trend from scipy.ndimage import uniform_filter1d rolling_mean = uniform_filter1d(data, size=7, mode='nearest') ax.plot(days, rolling_mean, color=color, linewidth=3, alpha=0.5, linestyle='--') # Confidence interval ax.fill_between(days, data-10, data+10, alpha=0.2, color=color) # Clinical targets ax.axhline(y=140, color='orange', linestyle=':', label='Stage 1 HTN') ax.axhline(y=130, color='green', linestyle=':', label='Target BP') # Styling ax.set_ylabel('SBP (mmHg)') ax.set_title(f'{group_name} (n=50)', fontsize=12, fontweight='bold', loc='left') ax.grid(True, alpha=0.3) ax.set_ylim(110, 180) # Add endpoint annotation endpoint_bp = data[-1] reduction = 160 - endpoint_bp ax.text(85, endpoint_bp, f'−{reduction:.0f} mmHg', bbox=dict(boxstyle='round', facecolor='white', edgecolor=color), fontsize=10, ha='right') # Plot each treatment arm # ======================= style_treatment_plot(ax1, placebo, 'gray', 'Placebo', 'Placebo Group') style_treatment_plot(ax2, drug_low_dose, 'blue', 'Drug 5mg', 'Low Dose (5mg)') style_treatment_plot(ax3, drug_high_dose, 'darkgreen', 'Drug 10mg', 'High Dose (10mg)') # Only show legend on first plot # ============================== ax1.legend(loc='upper right', fontsize=8) # X-axis label only on bottom # =========================== ax3.set_xlabel('Days Since Randomization', fontsize=12) # Add p-values and statistical significance # ========================================= fig.text(0.15, 0.88, 'Baseline: All groups ~160 mmHg', fontsize=10, style='italic') fig.text(0.15, 0.55, 'p < 0.01 vs placebo*', fontsize=10, color='blue') fig.text(0.15, 0.22, 'p < 0.001 vs placebo**', fontsize=10, color='darkgreen') # Main title # ========== fig.suptitle('REDUCE-BP Trial: Antihypertensive Efficacy by Treatment Arm\n' + 'Primary Endpoint: Change in Systolic BP at 90 Days', fontsize=16, fontweight='bold') plt.tight_layout(rect=[0, 0.02, 1, 0.96]) plt.show()

# Advanced pharmacokinetics visualization # ======================================= import numpy as np import matplotlib.pyplot as plt from matplotlib.gridspec import GridSpec from scipy import stats # Generate realistic drug concentration data # ========================================== np.random.seed(456) time_points = np.array([0, 0.25, 0.5, 1, 2, 3, 4, 6, 8, 12, 16, 20, 24]) # Two-compartment model with absorption # ===================================== def drug_concentration(t, dose=100, ka=2.0, ke=0.15, V=10): """Two-compartment PK model""" if t == 0: return 0 return (dose * ka / (V * (ka - ke))) * (np.exp(-ke * t) - np.exp(-ka * t)) # Generate concentrations with realistic variability # ================================================== theoretical_conc = [drug_concentration(t) for t in time_points] measured_conc = theoretical_conc + np.random.normal(0, 0.1*np.array(theoretical_conc)) measured_conc[measured_conc < 0] = 0 # Create complex figure layout # ============================ fig = plt.figure(figsize=(16, 12)) gs = GridSpec(4, 3, figure=fig, height_ratios=[2, 1, 1, 0.8], hspace=0.35, wspace=0.3) # Main PK curve # ============= ax_main = fig.add_subplot(gs[0, :]) ax_conc_dist = fig.add_subplot(gs[1, 0]) ax_elimination = fig.add_subplot(gs[1, 1]) ax_auc = fig.add_subplot(gs[1, 2]) ax_compartment = fig.add_subplot(gs[2, :2]) ax_clinical = fig.add_subplot(gs[2, 2]) ax_params = fig.add_subplot(gs[3, :]) # 1. Main concentration-time curve # ================================ time_smooth = np.linspace(0, 24, 500) theo_smooth = [drug_concentration(t) for t in time_smooth] # Plot with style # =============== ax_main.scatter(time_points[1:], measured_conc[1:], s=100, alpha=0.8, color='red', edgecolor='darkred', linewidth=2, label='Measured', zorder=5) ax_main.plot(time_smooth, theo_smooth, 'b-', linewidth=2.5, label='Model fit', alpha=0.8) # Add therapeutic window # ====================== ax_main.axhspan(2, 8, alpha=0.2, color='green', label='Therapeutic range') ax_main.axhline(y=1, color='orange', linestyle='--', linewidth=2, label='Minimum effective concentration') ax_main.axhline(y=10, color='red', linestyle='--', linewidth=2, label='Toxicity threshold') # Annotations # =========== tmax_idx = np.argmax(measured_conc) ax_main.annotate(f'Cmax = {measured_conc[tmax_idx]:.1f} μg/mL\nTmax = {time_points[tmax_idx]} h', xy=(time_points[tmax_idx], measured_conc[tmax_idx]), xytext=(time_points[tmax_idx]+2, measured_conc[tmax_idx]+1), arrowprops=dict(arrowstyle='->', color='black'), bbox=dict(boxstyle='round', facecolor='yellow', alpha=0.7)) ax_main.set_xlabel('Time (hours)', fontsize=12) ax_main.set_ylabel('Plasma Concentration (μg/mL)', fontsize=12) ax_main.set_title('Pharmacokinetic Profile - Single Oral Dose (100mg)', fontsize=14, fontweight='bold') ax_main.legend(loc='upper right') ax_main.grid(True, alpha=0.3) ax_main.set_xlim(-0.5, 24.5) ax_main.set_ylim(0, 12) # 2. Concentration distribution histogram # ======================================= ax_conc_dist.hist(measured_conc[1:], bins=8, orientation='horizontal', color='purple', alpha=0.7, edgecolor='black') ax_conc_dist.set_xlabel('Frequency') ax_conc_dist.set_ylabel('Concentration (μg/mL)') ax_conc_dist.set_title('Concentration Distribution', fontsize=11) ax_conc_dist.grid(True, alpha=0.3) # 3. Elimination rate # =================== elim_rate = -np.diff(measured_conc) / np.diff(time_points) time_mid = (time_points[1:] + time_points[:-1]) / 2 ax_elimination.bar(time_mid, elim_rate, width=0.8*np.diff(time_points), alpha=0.7, color='darkred', edgecolor='black') ax_elimination.set_xlabel('Time (hours)') ax_elimination.set_ylabel('Elimination Rate\n(μg/mL/h)') ax_elimination.set_title('Instantaneous Elimination Rate', fontsize=11) ax_elimination.grid(True, alpha=0.3) # 4. AUC visualization # ==================== ax_auc.fill_between(time_points, 0, measured_conc, alpha=0.5, color='lightblue', label='AUC₀₋₂₄') ax_auc.plot(time_points, measured_conc, 'b-', linewidth=2) ax_auc.set_xlabel('Time (hours)') ax_auc.set_ylabel('Concentration') ax_auc.set_title('Area Under Curve', fontsize=11) ax_auc.grid(True, alpha=0.3) # Calculate AUC using trapezoidal rule # ==================================== from scipy import integrate auc_value = integrate.trapz(measured_conc, time_points) ax_auc.text(0.5, 0.5, f'AUC = {auc_value:.1f}\nμg·h/mL', transform=ax_auc.transAxes, ha='center', fontsize=12, bbox=dict(boxstyle='round', facecolor='white')) # 5.
