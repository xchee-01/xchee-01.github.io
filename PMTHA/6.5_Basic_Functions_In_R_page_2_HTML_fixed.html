Missing Data Detection

# R Code: Lab results with missing values hemoglobin_levels <- c(12.5, 14.1, NA, 13.8, NA, 15.2, 11.9, NA) patient_ids <- c("P001", "P002", "P003", "P004", "P005", "P006", "P007", "P008") # Analyze missing data missing_count <- sum(is.na(hemoglobin_levels)) missing_percent <- round(missing_count / length(hemoglobin_levels) * 100, 1) # Identify patients needing retesting needs_retest <- patient_ids[is.na(hemoglobin_levels)] print("Patients needing repeat labs:") print(needs_retest) # Calculate stats on complete data valid_mean <- mean(hemoglobin_levels, na.rm = TRUE)

Kidney Function Assessment

# R Code: Calculate eGFR calculate_gfr <- function(age, weight_kg, creatinine_mg_dl, is_female = FALSE) { gfr <- ((140 - age) * weight_kg) / (72 * creatinine_mg_dl) if(is_female) { gfr <- gfr * 0.85 } return(round(gfr, 1)) } # Patient assessment gfr <- calculate_gfr(65, 70, 1.5, TRUE) print(paste("eGFR:", gfr, "mL/min")) if(gfr >= 90) { print("Stage 1: Normal kidney function") } else if(gfr >= 60) { print("Stage 2: Mild decrease in kidney function") } else if(gfr >= 30) { print("Stage 3: Moderate decrease - refer to nephrologist") } else { print("Stage 4-5: Severe decrease - urgent nephrology consult!") }
Copied!

Correct! GFR = ((140-65) * 70) / (72 * 1.5) * 0.85 = (75 * 70) / 108 * 0.85 = 48.6 * 0.85 = 41.3. Since 41.3 is between 30 and 60, it's Stage 3.
