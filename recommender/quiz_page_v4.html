// Global Variables const QUESTION_BANK_SHEET_ID = '10OvQxGnLMkieI3oBQcGrel0gzueoojjObZZQdWbs64o'; const RESULTS_SHEET_ID = '1IifIAO0odU4JFQHdSKllSkCo9wFgy2qMctqQ2dJQCwg'; const QUESTION_BANK_TAB = 'Prequiz_questions'; const RESULTS_TAB = 'prequiz_results'; // Handle GET requests with JSONP support function doGet(e) { try { // Log the incoming request for debugging Logger.log('doGet function called'); Logger.log('Event object: ' + JSON.stringify(e)); const params = e.parameter || {}; const action = params.action; const callback = params.callback; let response; // Handle data submission via GET (for no-cors mode) if (params.data) { const data = JSON.parse(decodeURIComponent(params.data)); if (data.action === 'saveBulkResults') { response = saveBulkResults(data.results); } } else { switch(action) { case 'getQuestions': response = getQuestions(); break; case 'debug': response = debugHeaders(); break; default: response = { error: 'Invalid action' }; } } // Return JSONP if callback is provided if (callback) { return ContentService.createTextOutput(callback + '(' + JSON.stringify(response) + ')') .setMimeType(ContentService.MimeType.JAVASCRIPT); } else { return ContentService.createTextOutput(JSON.stringify(response)) .setMimeType(ContentService.MimeType.JSON); } } catch (error) { Logger.log('Error in doGet: ' + error.toString()); Logger.log('Error stack: ' + error.stack); // Return error response const errorResponse = { success: false, error: error.toString() }; if (e && e.parameter && e.parameter.callback) { return ContentService.createTextOutput(e.parameter.callback + '(' + JSON.stringify(errorResponse) + ')') .setMimeType(ContentService.MimeType.JAVASCRIPT); } return ContentService.createTextOutput(JSON.stringify(errorResponse)) .setMimeType(ContentService.MimeType.JSON); } } // Debug function to see headers and data structure function debugHeaders() { try { const questionBankSheet = SpreadsheetApp.openById(QUESTION_BANK_SHEET_ID).getSheetByName(QUESTION_BANK_TAB); const data = questionBankSheet.getDataRange().getValues(); const headers = data[0]; const firstRow = data[1]; return { success: true, headers: headers, firstRow: firstRow, totalRows: data.length }; } catch (error) { Logger.log('Error in debugHeaders: ' + error.toString()); return { success: false, error: error.toString() }; } } // Get all questions from the question bank function getQuestions() { try { const questionBankSheet = SpreadsheetApp.openById(QUESTION_BANK_SHEET_ID).getSheetByName(QUESTION_BANK_TAB); const data = questionBankSheet.getDataRange().getValues(); const headers = data[0]; // Map column headers to indices (handle case sensitivity and whitespace) const columnMap = {}; headers.forEach((header, index) => { // Normalize header by trimming and making case-insensitive const normalizedHeader = header.toString().trim(); columnMap[normalizedHeader] = index; // Also create versions without spaces for flexibility const noSpaceHeader = normalizedHeader.replace(/\s+/g, ''); columnMap[noSpaceHeader] = index; // Log for debugging Logger.log(`Header ${index}: "${header}" -> "${normalizedHeader}"`); }); // Log column map for debugging Logger.log('Column map: ' + JSON.stringify(columnMap)); // Process questions into JSON format const questions = []; for (let i = 1; i < data.length; i++) { const row = data[i]; // Skip empty rows if (!row[0]) continue; // Try different header variations to find the right column const getColumn = (variations) => { for (let v of variations) { if (columnMap[v] !== undefined) { return columnMap[v]; } } Logger.log('Could not find column for variations: ' + variations.join(', ')); return -1; }; // Define column variations to try const questionIdCol = getColumn(['Question ID', 'QuestionID', 'question id']); const questionTypeCol = getColumn(['Question Type', 'QuestionType', 'question type']); const questionTextCol = getColumn(['Question Text', 'QuestionText', 'question text']); const optionsCol = getColumn(['Options', 'options']); const correctAnswerCol = getColumn(['Correct Answer', 'CorrectAnswer', 'correct answer']); const explanationCol = getColumn(['Explanation', 'explanation']); const hintsCol = getColumn(['Hints', 'hints']); const codeSnippetsCol = getColumn(['Code Snippets', 'CodeSnippets', 'code snippets']); const mediaLinksCol = getColumn(['Media links', 'MediaLinks', 'media links']); const learningConceptsCol = getColumn(['Learning concepts', 'LearningConcepts', 'learning concepts']); const difficultyLevelCol = getColumn(['Difficulty Level', 'DifficultyLevel', 'difficulty level']); const bloomsLevelCol = getColumn(['Bloom\'s Level', 'Bloom\'sLevel', 'Blooms Level', 'bloom\'s level']); // Log the values being processed for the first few rows for debugging if (i <= 3) { Logger.log(`Processing row ${i}:`); Logger.log(` Question ID: ${row[questionIdCol]}`); Logger.log(` Options: ${row[optionsCol]}`); Logger.log(` Correct Answer: ${row[correctAnswerCol]}`); } // Parse options (semicolon separated) let options = []; if (optionsCol >= 0 && row[optionsCol]) { const optionsString = row[optionsCol].toString(); options = optionsString.split(';').map(opt => opt.trim()).filter(opt => opt); // Log parsed options for debugging if (i <= 3) { Logger.log(` Parsed options: ${JSON.stringify(options)}`); } } // Parse learning concepts (semicolon separated) let concepts = []; if (learningConceptsCol >= 0 && row[learningConceptsCol]) { const conceptsString = row[learningConceptsCol].toString(); concepts = conceptsString.split(';').map(concept => concept.trim()).filter(concept => concept); } questions.push({ questionID: questionIdCol >= 0 ? row[questionIdCol] : '', questionType: questionTypeCol >= 0 ? row[questionTypeCol] : '', questionText: questionTextCol >= 0 ? row[questionTextCol] : '', options: options, correctAnswer: correctAnswerCol >= 0 ? row[correctAnswerCol] : '', explanation: explanationCol >= 0 ? row[explanationCol] : '', hints: hintsCol >= 0 ? row[hintsCol] : '', codeSnippets: codeSnippetsCol >= 0 ? row[codeSnippetsCol] : '', mediaLinks: mediaLinksCol >= 0 ? row[mediaLinksCol] : '', learningConcepts: concepts, difficultyLevel: difficultyLevelCol >= 0 ? row[difficultyLevelCol] : '', bloomsLevel: bloomsLevelCol >= 0 ? row[bloomsLevelCol] : '' }); } Logger.log(`Successfully processed ${questions.length} questions`); return { success: true, data: questions }; } catch (error) { Logger.log('Error in getQuestions: ' + error.toString()); Logger.log('Error stack: ' + error.stack); return { success: false, error: error.toString() }; } } // Save all results at once (for bulk submission) function saveBulkResults(results) { try { const resultsSheet = SpreadsheetApp.openById(RESULTS_SHEET_ID); let resultsTab = resultsSheet.getSheetByName(RESULTS_TAB); // Create the results tab if it doesn't exist if (!resultsTab) { resultsTab = resultsSheet.insertSheet(RESULTS_TAB); // Set up headers const headers = [ 'Time Stamp', 'Question ID', 'Question Type', 'Question Text', 'User Selected Option', 'Correct Answer', 'Is Correct? (Yes or No)', 'Learning concepts', 'Difficulty Level', 'Bloom\'s Level' ]; resultsTab.getRange(1, 1, 1, headers.length).setValues([headers]); resultsTab.getRange(1, 1, 1, headers.length).setFontWeight('bold'); resultsTab.setFrozenRows(1); } // Get the next available row const lastRow = resultsTab.getLastRow(); const nextRow = lastRow + 1; // Prepare data for insertion const rowsData = results.map(result => [ new Date(), // Time Stamp result.questionID, result.questionType, result.questionText, result.userAnswer, result.correctAnswer, result.isCorrect ? 'Yes' : 'No', Array.isArray(result.learningConcepts) ? result.learningConcepts.join('; ') : result.learningConcepts, result.difficultyLevel, result.bloomsLevel ]); // Insert the data if (rowsData.length > 0) { resultsTab.getRange(nextRow, 1, rowsData.length, rowsData[0].length).setValues(rowsData); } Logger.log(`Successfully saved ${rowsData.length} results`); return { success: true }; } catch (error) { Logger.log('Error in saveBulkResults: ' + error.toString()); Logger.log('Error stack: ' + error.stack); return { success: false, error: error.toString() }; } } // Save individual result (for question-by-question submission) function saveResult(quizResult) { try { const resultsSheet = SpreadsheetApp.openById(RESULTS_SHEET_ID); let resultsTab = resultsSheet.getSheetByName(RESULTS_TAB); // Create the results tab if it doesn't exist if (!resultsTab) { resultsTab = resultsSheet.insertSheet(RESULTS_TAB); // Set up headers const headers = [ 'Time Stamp', 'Question ID', 'Question Type', 'Question Text', 'User Selected Option', 'Correct Answer', 'Is Correct? (Yes or No)', 'Learning concepts', 'Difficulty Level', 'Bloom\'s Level' ]; resultsTab.getRange(1, 1, 1, headers.length).setValues([headers]); resultsTab.getRange(1, 1, 1, headers.length).setFontWeight('bold'); resultsTab.setFrozenRows(1); } // Get the next available row const lastRow = resultsTab.getLastRow(); const nextRow = lastRow + 1; // Prepare data for insertion const rowData = [ new Date(), // Time Stamp quizResult.questionID, quizResult.questionType, quizResult.questionText, quizResult.userAnswer, quizResult.correctAnswer, quizResult.isCorrect ? 'Yes' : 'No', Array.isArray(quizResult.learningConcepts) ? quizResult.learningConcepts.join('; ') : quizResult.learningConcepts, quizResult.difficultyLevel, quizResult.bloomsLevel ]; // Insert the data resultsTab.getRange(nextRow, 1, 1, rowData.length).setValues([rowData]); return { success: true }; } catch (error) { Logger.log('Error in saveResult: ' + error.toString()); return { success: false, error: error.toString() }; } } // Test functions for debugging (you can run these directly from the Google Apps Script editor) function testGetQuestions() { // Test the getQuestions function directly console.log('Starting testGetQuestions...');
