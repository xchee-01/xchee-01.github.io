Different colormaps serve different scientific purposes. 'Viridis' is excellent for general use, while 'Plasma' provides high contrast for highlighting changes. 'Inferno' and 'Magma' are great for thermal data, and 'Cividis' remains interpretable for people with color vision deficiencies.

Creating Custom Scientific Colormaps

# Creating a custom colormap for ocean depth # Blue gradient from light (surface) to dark (deep) colors = [(0.8, 0.95, 1), (0.6, 0.8, 1), (0.4, 0.6, 0.8), (0.2, 0.4, 0.6), (0.1, 0.2, 0.4), (0, 0.1, 0.2)] ocean_cmap = LinearSegmentedColormap.from_list('ocean_depth', colors) # Generate synthetic ocean depth data depth = np.zeros((100, 100)) # Create a continental shelf and deep ocean for i in range(100): for j in range(100): # Distance from top-left corner d = np.sqrt((i - 10)**2 + (j - 10)**2) if d < 30: # Continental shelf depth[i, j] = 200 + d * 10 # Gradually deepening elif d < 50: # Continental slope depth[i, j] = 500 + (d - 30) * 70 # Rapidly deepening else: # Deep ocean depth[i, j] = 1900 + (d - 50) * 20 # Gradually deepening # Visualize the ocean depth with our custom colormap fig, ax = plt.subplots(figsize=(10, 8)) im = ax.imshow(depth, cmap=ocean_cmap) cbar = fig.colorbar(im, ax=ax) cbar.set_label('Ocean Depth (m)', rotation=270, labelpad=20) ax.set_title('Ocean Bathymetry Visualization') # Add contour lines to enhance depth perception contours = ax.contour(depth, colors='white', alpha=0.5, levels=10) ax.clabel(contours, inline=True, fontsize=8, fmt='%1.0f m') plt.show()

# Basic 3D surface plot: Enzyme catalytic activity as a function of temperature and pH fig = plt.figure(figsize=(10, 8)) ax = fig.add_subplot(111, projection='3d') # Create 3D axes # Create a grid of temperature and pH values temp = np.linspace(10, 70, 30) # Temperature from 10°C to 70°C pH = np.linspace(3, 10, 30) # pH from 3 to 10 temp_grid, pH_grid = np.meshgrid(temp, pH) # Calculate enzyme activity based on temperature and pH # Model: Bell curve for temperature optimum, bell curve for pH optimum temp_effect = np.exp(-0.01 * (temp_grid - 37)**2) # Optimum at 37°C pH_effect = np.exp(-1.0 * (pH_grid - 7)**2) # Optimum at pH 7 activity = 100 * temp_effect * pH_effect # Combined effect (max activity = 100) # Create the surface plot surf = ax.plot_surface(temp_grid, pH_grid, activity, cmap='viridis', edgecolor='none', alpha=0.9) # Add labels and title ax.set_xlabel('Temperature (°C)') ax.set_ylabel('pH') ax.set_zlabel('Enzyme Activity (%)') ax.set_title('Enzyme Activity as a Function of Temperature and pH') # Add a color bar fig.colorbar(surf, ax=ax, shrink=0.5, aspect=5, label='Activity (%)') plt.tight_layout() plt.show()

3D Contour Plots and Vector Fields

# 3D contour plot with vector field: Visualizing an electric field fig = plt.figure(figsize=(12, 10)) ax = fig.add_subplot(111, projection='3d') # Create a grid in 3D space x = np.linspace(-5, 5, 10) y = np.linspace(-5, 5, 10) z = np.linspace(-5, 5, 10) x_grid, y_grid, z_grid = np.meshgrid(x, y, z) # Calculate electric potential: point charge at origin # V = k*q/r where r is distance from origin r = np.sqrt(x_grid**2 + y_grid**2 + z_grid**2) # Avoid division by zero at origin r = np.where(r < 0.1, 0.1, r) V = 1/r # Electric potential (arbitrary units) # Calculate electric field components (E = -∇V) Ex, Ey, Ez = np.gradient(-V, *[x[1]-x[0] for x in [x, y, z]]) # Plot the contour surfaces of the potential contour_levels = [0.2, 0.3, 0.5, 1.0] ax.contour(x_grid[:,:,5], y_grid[:,:,5], V[:,:,5], levels=contour_levels, cmap='Blues', alpha=0.6) # Plot vector field (electric field) # We'll use a slice of the 3D grid for clarity x_slice = x_grid[:,:,5] y_slice = y_grid[:,:,5] z_slice = np.zeros_like(x_slice) + z[5] # Fixed z value for the slice Ex_slice = Ex[:,:,5] Ey_slice = Ey[:,:,5] Ez_slice = Ez[:,:,5] # Normalize the field vectors for better visualization E_magnitude = np.sqrt(Ex_slice**2 + Ey_slice**2 + Ez_slice**2) scale_factor = 0.5 / np.max(E_magnitude) Ex_slice = Ex_slice * scale_factor Ey_slice = Ey_slice * scale_factor Ez_slice = Ez_slice * scale_factor # Plot the vector field ax.quiver(x_slice, y_slice, z_slice, Ex_slice, Ey_slice, Ez_slice, color='red', length=1, normalize=False, alpha=0.7) # Add labels and title ax.set_xlabel('X (m)') ax.set_ylabel('Y (m)') ax.set_zlabel('Z (m)') ax.set_title('Electric Potential and Field of a Point Charge') # Add a text explanation ax.text2D(0.05, 0.95, "Electric Field Visualization\n" "- Blue surfaces: Equipotential surfaces\n" "- Red arrows: Electric field vectors\n" "- Field strength decreases with distance\n" "- Field direction is radially outward", transform=ax.transAxes, fontsize=10, bbox=dict(facecolor='white', alpha=0.8)) # Set the view angle ax.view_init(elev=30, azim=45) plt.tight_layout() plt.show()

This advanced 3D visualization combines contour surfaces showing equipotential lines with vector arrows representing the electric field of a point charge. This type of plot is invaluable in physics education and research to help students and scientists visualize invisible electromagnetic fields and understand how they relate to electric potential.
