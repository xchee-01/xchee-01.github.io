// This code goes in Google Apps Script // You'll need to create a new Google Sheet with columns for email and password // Replace this with your actual Google Sheet ID // You can find this in the URL of your sheet: https://docs.google.com/spreadsheets/d/SHEET_ID_HERE/edit const SHEET_ID = 'YOUR_GOOGLE_SHEET_ID'; const SHEET_NAME = 'Users'; // The name of the sheet tab function doGet(e) { // Handle GET requests (not used in this example) return ContentService.createTextOutput(JSON.stringify({ status: 'error', message: 'This API only accepts POST requests' })).setMimeType(ContentService.MimeType.JSON); } function doPost(e) { try { // Parse the request body const data = JSON.parse(e.postData.contents); const action = data.action; if (action === 'login') { return handleLogin(data); } else if (action === 'register') { return handleRegistration(data); } else { return ContentService.createTextOutput(JSON.stringify({ status: 'error', message: 'Unknown action' })).setMimeType(ContentService.MimeType.JSON); } } catch (error) { return ContentService.createTextOutput(JSON.stringify({ status: 'error', message: 'Invalid request: ' + error.toString() })).setMimeType(ContentService.MimeType.JSON); } } function handleLogin(data) { const email = data.email; const password = data.password; // In a real application, you would NEVER store plain text passwords // This is just for demonstration purposes // Passwords should be hashed and salted const sheet = SpreadsheetApp.openById(SHEET_ID).getSheetByName(SHEET_NAME); const values = sheet.getDataRange().getValues(); // Skip header row if you have one for (let i = 1; i < values.length; i++) { if (values[i][0] === email && values[i][1] === password) { return ContentService.createTextOutput(JSON.stringify({ success: true, message: 'Login successful' })).setMimeType(ContentService.MimeType.JSON); } } return ContentService.createTextOutput(JSON.stringify({ success: false, message: 'Invalid email or password' })).setMimeType(ContentService.MimeType.JSON); } function handleRegistration(data) { const email = data.email; const password = data.password; // Check if user already exists const sheet = SpreadsheetApp.openById(SHEET_ID).getSheetByName(SHEET_NAME); const values = sheet.getDataRange().getValues(); for (let i = 1; i < values.length; i++) { if (values[i][0] === email) { return ContentService.createTextOutput(JSON.stringify({ success: false, message: 'Email already registered' })).setMimeType(ContentService.MimeType.JSON); } } // Add new user sheet.appendRow([email, password]); return ContentService.createTextOutput(JSON.stringify({ success: true, message: 'Registration successful' })).setMimeType(ContentService.MimeType.JSON); }
